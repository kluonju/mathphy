\section{拉普拉斯反变换}

\begin{frame}{推广的约当引理}
  \textbf{推广的约当引理}。Bromwich 积分
  \begin{equation*}
    f(t)=\frac{1}{2\pi\mathrm{i}}
      \int_{\sigma-\mathrm{i}\infty}^{\sigma+\mathrm{i}\infty}
      \bar f(p)\,e^{pt}\,\mathrm{d}p.
  \end{equation*}
  像函数是 $p$ 的解析函数，积分可以通过\textbf{留数定理}求得。
  \vspace{0.3em}

  设 $C_R$ 是以原点为圆心、半径为 $R$ 的圆周在直线 $\mathrm{Re}\,p=a>0$ 左侧的圆弧。
  若 $|p|\to\infty$ 时 $\bar f(p)$ 在这段辐角范围内一致趋于零，则
  \begin{equation*}
    \lim_{R\to\infty}\int_{C_R}\bar f(p)\,e^{pt}\,\mathrm{d}p=0
    \qquad(t>0).
  \end{equation*}
\end{frame}

\begin{frame}{留数之和}
  所作路径在 $\mathrm{Re}\,p=a$ 的左半平面。$t>0$ 时把竖直线用左圆弧闭上，圆弧上的积分趋于零，
  \begin{equation*}
    f(t)=\sum_j\mathrm{Res}\bigl[\bar f(p)\,e^{pt},\,p_j\bigr].
  \end{equation*}
  留数取闭路所围的奇点。简单极点 $p_j$ 上，
  \begin{equation*}
    \mathrm{Res}
    =\lim_{p\to p_j}(p-p_j)\,\bar f(p)\,e^{pt}.
  \end{equation*}
  $n$ 阶极点：
  \begin{equation*}
    \mathrm{Res}
    =\frac{1}{(n-1)!}
      \lim_{p\to p_j}
      \frac{\mathrm{d}^{n-1}}{\mathrm{d}p^{n-1}}
      \bigl[(p-p_j)^n\bar f(p)\,e^{pt}\bigr].
  \end{equation*}
\end{frame}

\begin{frame}{留数反演：$\sin\omega t$}
  取 $\bar f(p)=\omega/(p^2+\omega^2)$，$\omega>0$。极点
  $p=\pm\mathrm{i}\omega$ 都在左闭路内（Bromwich 线在右侧）。
  \begin{align*}
    \mathrm{Res}_{p=\mathrm{i}\omega}
    &=\lim_{p\to\mathrm{i}\omega}
      (p-\mathrm{i}\omega)\,
      \frac{\omega\,e^{pt}}{(p-\mathrm{i}\omega)(p+\mathrm{i}\omega)}
    =\frac{\omega\,e^{\mathrm{i}\omega t}}{2\mathrm{i}\omega}
    =\frac{e^{\mathrm{i}\omega t}}{2\mathrm{i}},\\[0.35em]
    \mathrm{Res}_{p=-\mathrm{i}\omega}
    &=\frac{\omega\,e^{-\mathrm{i}\omega t}}{-2\mathrm{i}\omega}
    =-\frac{e^{-\mathrm{i}\omega t}}{2\mathrm{i}}.
  \end{align*}
  相加：
  \begin{equation*}
    f(t)
    =\frac{e^{\mathrm{i}\omega t}-e^{-\mathrm{i}\omega t}}{2\mathrm{i}}
    =\sin\omega t.
  \end{equation*}
\end{frame}

\begin{frame}{留数反演：$\cos\omega t$}
  取 $\bar f(p)=p/(p^2+\omega^2)$。同样在 $\pm\mathrm{i}\omega$，
  \begin{align*}
    \mathrm{Res}_{p=\mathrm{i}\omega}
    &=\frac{\mathrm{i}\omega\,e^{\mathrm{i}\omega t}}{2\mathrm{i}\omega}
    =\frac{e^{\mathrm{i}\omega t}}{2},\\[0.35em]
    \mathrm{Res}_{p=-\mathrm{i}\omega}
    &=\frac{(-\mathrm{i}\omega)\,e^{-\mathrm{i}\omega t}}{-2\mathrm{i}\omega}
    =\frac{e^{-\mathrm{i}\omega t}}{2}.
  \end{align*}
  相加得 $\cos\omega t$。与查表一致，但步骤只依赖留数公式。
\end{frame}

\begin{frame}{留数反演：$1/[p(p+a)]$}
  $a>0$。极点 $p=0$（一阶）与 $p=-a$（一阶）。
  \begin{align*}
    \mathrm{Res}_{p=0}
    &=\lim_{p\to 0}p\cdot\frac{e^{pt}}{p(p+a)}
    =\frac{1}{a},\\[0.35em]
    \mathrm{Res}_{p=-a}
    &=\lim_{p\to -a}(p+a)\cdot\frac{e^{pt}}{p(p+a)}
    =\frac{e^{-at}}{-a}.
  \end{align*}
  因此
  \begin{equation*}
    f(t)=\frac{1-e^{-at}}{a}.
  \end{equation*}
  用裂项 $1/[p(p+a)]=(1/a)(1/p-1/(p+a))$ 再逐项反演，得到同一结果。
\end{frame}

\begin{frame}{留数反演：二阶极点}
  取 $\bar f(p)=1/(p+\alpha)^2$，$\alpha$ 实。$p=-\alpha$ 是二阶极点，
  \begin{align*}
    \mathrm{Res}
    &=\lim_{p\to-\alpha}
      \frac{\mathrm{d}}{\mathrm{d}p}
      \bigl[(p+\alpha)^2\cdot\tfrac{e^{pt}}{(p+\alpha)^2}\bigr]
    =\lim_{p\to-\alpha}\frac{\mathrm{d}}{\mathrm{d}p}\,e^{pt}
    =t\,e^{-\alpha t}.
  \end{align*}
  故 $\mathcal{L}^{-1}\{1/(p+\alpha)^2\}=t e^{-\alpha t}$，
  与位移定理作用于 $\mathcal{L}\{t\}=1/p^2$ 一致。
\end{frame}

\begin{frame}{像函数对 $p$ 求导}
  $f(t)$ 满足拉普拉斯变换的充分条件时，$\bar f(p)$ 在 $\mathrm{Re}\,p>\sigma_0$ 的半平面解析，可以在积分号下求导：
  \begin{align*}
    \bar f^{(n)}(p)
    &=\frac{\mathrm{d}^n}{\mathrm{d}p^n}
      \int_0^{\infty} f(t)\,e^{-pt}\,\mathrm{d}t \\
    &=\int_0^{\infty}(-t)^n f(t)\,e^{-pt}\,\mathrm{d}t.
  \end{align*}
  所以
  \begin{equation*}
    \bar f^{(n)}(p)\risingdotseq (-t)^n f(t).
  \end{equation*}
  特别地，$\mathcal{L}\{1\}=1/p$，
  \begin{equation*}
    \frac{1}{p^2}=-\frac{\mathrm{d}}{\mathrm{d}p}\frac{1}{p}\risingdotseq t,
    \qquad
    \frac{1}{p^3}=\frac12\frac{\mathrm{d}^2}{\mathrm{d}p^2}\frac{1}{p}
      \risingdotseq\frac12 t^2.
  \end{equation*}
\end{frame}

\begin{frame}{例：$t\sin$ 与 $t\cos$}
  $\sin\omega t\fallingdotseq\omega/(p^2+\omega^2)$。对 $p$ 求导，
  \begin{equation*}
    \frac{\mathrm{d}}{\mathrm{d}p}\frac{\omega}{p^2+\omega^2}
    =-\frac{2p\omega}{(p^2+\omega^2)^2}
    \risingdotseq -t\sin\omega t.
  \end{equation*}
  因此
  \begin{equation*}
    t\sin\omega t
    \fallingdotseq\frac{2p\omega}{(p^2+\omega^2)^2},
    \qquad
    t\cos\omega t
    \fallingdotseq\frac{p^2-\omega^2}{(p^2+\omega^2)^2}.
  \end{equation*}
  第二式同样来自 $\cos\omega t\fallingdotseq p/(p^2+\omega^2)$ 对 $p$ 的导数。
\end{frame}

\begin{frame}{待定系数}
  把
  \begin{equation*}
    \frac{1}{p^3(p+\alpha)}
    =\frac{A}{p^3}+\frac{B}{p^2}+\frac{C}{p}+\frac{D}{p+\alpha}
  \end{equation*}
  通分。分子
  \begin{equation*}
    A(p+\alpha)+Bp(p+\alpha)+Cp^2(p+\alpha)+Dp^3=1.
  \end{equation*}
  $p=0$ 给出 $A\alpha=1$，故 $A=1/\alpha$。
  $p=-\alpha$ 给出 $D(-\alpha)^3=1$，故 $D=-1/\alpha^3$。
  $p^3$ 的系数 $C+D=0$，故 $C=1/\alpha^3$。
  $p^2$ 的系数 $B+C\alpha=0$，故 $B=-1/\alpha^2$。
\end{frame}

\begin{frame}{裂项之后逐项反演}
  \begin{equation*}
    \frac{1}{p^3(p+\alpha)}
    =\frac{1}{\alpha}\frac{1}{p^3}
     -\frac{1}{\alpha^2}\frac{1}{p^2}
     +\frac{1}{\alpha^3}\frac{1}{p}
     -\frac{1}{\alpha^3}\frac{1}{p+\alpha}.
  \end{equation*}
  $1/p^3\risingdotseq t^2/2$，$1/p^2\risingdotseq t$，$1/p\risingdotseq 1$，
  $1/(p+\alpha)\risingdotseq e^{-\alpha t}$。因此原函数是
  \begin{equation*}
    \frac{t^2}{2\alpha}-\frac{t}{\alpha^2}
    +\frac{1}{\alpha^3}-\frac{e^{-\alpha t}}{\alpha^3}.
  \end{equation*}
  $1/p^2$ 前面的系数是负的，所以 $t$ 的系数是 $-1/\alpha^2$。
\end{frame}

\begin{frame}{支点：$1/\sqrt{p}$}
  被积函数 $e^{pt}/\sqrt{p}$ 的唯一奇点是 $p=0$，这是极点型支点。
  自 $p=0$ 沿负实轴把平面切开，再在一个单值分支上积分。
  大圆弧 $C_R$ 和小圆 $C_\varepsilon$ 都以原点为圆心。
  \vspace{0.3em}

  令 $R\to\infty$，由推广的约当引理，$\displaystyle\int_{C_R}\to 0$。
  小圆上 $|e^{pt}/\sqrt{p}|$ 乘上弧长 $2\pi\varepsilon$，在 $\varepsilon\to 0$ 时也趋于零。
  闭路内无奇点，竖直线上的积分等于割线两侧积分的相反数。
\end{frame}

\begin{frame}{沿割线的辐角}
  割线下沿的辐角为 $-\pi$，上沿的辐角为 $\pi$。于是
  \begin{equation*}
    \sqrt{p}=\begin{cases}
      -\mathrm{i}\sqrt{|p|}, & \text{下沿},\\
      \mathrm{i}\sqrt{|p|}, & \text{上沿}.
    \end{cases}
  \end{equation*}
  Bromwich 积分等于
  \begin{equation*}
    f(t)=-\frac{1}{2\pi\mathrm{i}}
      \Biggl(
        \int_0^{-\infty}\frac{e^{pt}}{-\mathrm{i}\sqrt{|p|}}\,\mathrm{d}p
        +\int_{-\infty}^{0}\frac{e^{pt}}{\mathrm{i}\sqrt{|p|}}\,\mathrm{d}p
      \Biggr).
  \end{equation*}
  $1/(-\mathrm{i})=\mathrm{i}$，$1/\mathrm{i}=-\mathrm{i}$。两项化简后同号，
  \begin{equation*}
    f(t)=\frac{1}{\pi}\int_{-\infty}^{0}
      \frac{e^{pt}}{\sqrt{|p|}}\,\mathrm{d}p.
  \end{equation*}
\end{frame}

\begin{frame}{换成高斯积分}
  令 $y=\sqrt{|p|\,t}$。$p$ 从 $-\infty$ 到 $0$ 时，$y$ 从 $\infty$ 到 $0$，
  $p=-y^2/t$，$\mathrm{d}p=-(2y/t)\,\mathrm{d}y$，$1/\sqrt{|p|}=\sqrt{t}/y$。
  \begin{align*}
    f(t)
    &=\frac{1}{\pi}\int_{\infty}^{0}
      e^{-y^2}\,\frac{\sqrt{t}}{y}
      \Bigl(-\frac{2y}{t}\Bigr)\mathrm{d}y \\
    &=\frac{2}{\pi\sqrt{t}}\int_0^{\infty} e^{-y^2}\,\mathrm{d}y
     =\frac{2}{\pi\sqrt{t}}\cdot\frac{\sqrt{\pi}}{2}
     =\frac{1}{\sqrt{\pi t}}.
  \end{align*}
  反过来说，$\mathcal{L}\{1/\sqrt{\pi t}\}=1/\sqrt{p}$。
\end{frame}

\begin{frame}{例：延迟与卷积}
  求 $e^{-\alpha p}/[p(p+b)]$ 的原函数。
  $1/p\risingdotseq\Theta(t)$。应用延迟定理，
  \begin{equation*}
    \frac{e^{-\alpha p}}{p}\risingdotseq\Theta(t-\alpha),
    \qquad
    \frac{1}{p+b}\risingdotseq e^{-bt}.
  \end{equation*}
  像函数的乘积对应卷积，
  \begin{align*}
    \int_0^t\Theta(\tau-\alpha)\,e^{-b(t-\tau)}\,\mathrm{d}\tau
    &=\Theta(t-\alpha)\int_{\alpha}^{t} e^{-b(t-\tau)}\,\mathrm{d}\tau \\
    &=\Theta(t-\alpha)
      \Biggl[\frac{1}{b}e^{-b(t-\tau)}\Biggr]_{\alpha}^{t} \\
    &=\frac{1-e^{-b(t-\alpha)}}{b}\,\Theta(t-\alpha).
  \end{align*}
  $t<\alpha$ 时结果为零。也可以先把分母裂成 $1/b\,(1/p-1/(p+b))$，再对每一项用延迟定理。
\end{frame}

\section{积分变换的应用}

\begin{frame}{计算 $\sum 1/n^2$}
  由 $\displaystyle\int_0^{\infty} t\,e^{-pt}\,\mathrm{d}t=1/p^2$（$\mathrm{Re}\,p>0$），取 $p=n$，
  \begin{equation*}
    \frac{1}{n^2}=\int_0^{\infty} t\,e^{-nt}\,\mathrm{d}t.
  \end{equation*}
  求和，
  \begin{equation*}
    \sum_{n=1}^{\infty}\frac{1}{n^2}
    =\int_0^{\infty} t\Biggl(\sum_{n=1}^{\infty} e^{-nt}\Biggr)\mathrm{d}t.
  \end{equation*}
  括号里是公比为 $e^{-t}$ 的等比级数，
  \begin{equation*}
    \sum_{n=1}^{\infty} e^{-nt}
    =\frac{e^{-t}}{1-e^{-t}}
    =\frac{1}{e^{t}-1}.
  \end{equation*}
  因此
  \begin{equation*}
    \sum_{n=1}^{\infty}\frac{1}{n^2}
    =\int_0^{\infty}\frac{t}{e^{t}-1}\,\mathrm{d}t
    =\frac{\pi^2}{6}.
  \end{equation*}
\end{frame}

\begin{frame}{用 Parseval 等式核对}
  在 $[-\pi,\pi]$ 上取 $f(x)=x$，则 $\ell=\pi$，
  \begin{equation*}
    c_n=\frac{1}{2\pi}\int_{-\pi}^{\pi} x\,e^{-\mathrm{i} nx}\,\mathrm{d}x.
  \end{equation*}
  分部：$u=x$，$v=e^{-\mathrm{i} nx}/(-\mathrm{i} n)$（$n\neq 0$）。
  $\displaystyle\int e^{-\mathrm{i} nx}\,\mathrm{d}x$ 在整周期上为零，只剩边界。
  $e^{-\mathrm{i} n\pi}=e^{\mathrm{i} n\pi}=(-1)^n$，
  \begin{align*}
    \Biggl[\frac{x\,e^{-\mathrm{i} nx}}{-\mathrm{i} n}\Biggr]_{-\pi}^{\pi}
    &=\frac{\pi(-1)^n}{-\mathrm{i} n}
      -\frac{(-\pi)(-1)^n}{-\mathrm{i} n}
     =\frac{2\pi\,\mathrm{i}}{n}(-1)^n.
  \end{align*}
  这里用了 $1/(-\mathrm{i})=\mathrm{i}$。再除以 $2\pi$，
  \begin{equation*}
    c_n=\frac{(-1)^n}{n}\,\mathrm{i},
    \qquad |c_n|^2=\frac{1}{n^2},
    \qquad c_0=0.
  \end{equation*}
\end{frame}

\begin{frame}{核对的后一半}
  Parseval 等式
  \begin{equation*}
    \sum_{n=-\infty}^{\infty}|c_n|^2
    =\frac{1}{2\pi}\int_{-\pi}^{\pi} x^2\,\mathrm{d}x.
  \end{equation*}
  左边是 $2\sum_{n=1}^{\infty} 1/n^2$。右边
  \begin{equation*}
    \frac{1}{2\pi}\cdot\frac{2\pi^3}{3}=\frac{\pi^2}{3}.
  \end{equation*}
  所以
  \begin{equation*}
    \sum_{n=1}^{\infty}\frac{1}{n^2}=\frac{\pi^2}{6}.
  \end{equation*}
  与拉普拉斯变换把级数改成 $\displaystyle\int_0^{\infty} t/(e^t-1)\,\mathrm{d}t$ 的结果一致。
\end{frame}

\begin{frame}{含 $1/t$ 的像}
  若 $\displaystyle\int_p^{\infty}\bar f(q)\,\mathrm{d}q$ 存在，且 $t\to 0$ 时 $|f(t)/t|$ 有界，则
  \begin{equation*}
    \int_p^{\infty}\bar f(q)\,\mathrm{d}q
    \risingdotseq\frac{f(t)}{t}.
  \end{equation*}
  这是对参数 $p$ 的积分：$\displaystyle\int_p^{\infty}\mathcal{L}\{f\}\,\mathrm{d}q
    =\int_0^{\infty} f(t)\bigl(\int_p^{\infty} e^{-qt}\,\mathrm{d}q\bigr)\mathrm{d}t
    =\mathcal{L}\{f(t)/t\}$。
  \vspace{0.3em}

  再令 $p\to 0$，
  \begin{equation*}
    \int_0^{\infty}\frac{f(t)}{t}\,\mathrm{d}t
    =\int_0^{\infty}\bar f(p)\,\mathrm{d}p.
  \end{equation*}
\end{frame}

\begin{frame}{例：$(\sin\omega t)/t$}
  $\mathcal{L}\{\sin\omega t\}=\omega/(p^2+\omega^2)$。对像函数从 $p$ 积到 $\infty$，
  \begin{align*}
    \int_p^{\infty}\frac{\omega}{q^2+\omega^2}\,\mathrm{d}q
    &=\bigl[\arctan(q/\omega)\bigr]_p^{\infty}
     =\frac{\pi}{2}-\arctan\frac{p}{\omega}.
  \end{align*}
  因此
  \begin{equation*}
    \frac{\sin\omega t}{t}
    \fallingdotseq\frac{\pi}{2}-\arctan\frac{p}{\omega}.
  \end{equation*}
  令 $p\to 0$、$\omega=1$，
  \begin{equation*}
    \int_0^{\infty}\frac{\sin t}{t}\,\mathrm{d}t
    =\int_0^{\infty}\frac{\mathrm{d}p}{p^2+1}
    =\frac{\pi}{2}.
  \end{equation*}
\end{frame}

\begin{frame}{例：两个余弦之差}
  $a>0$，$b>0$。$\mathcal{L}\{\cos at\}=p/(p^2+a^2)$，
  \begin{align*}
    &\int_0^{\infty}\frac{\cos(at)-\cos(bt)}{t}\,\mathrm{d}t \\
    &\quad=\int_0^{\infty}
      \Biggl(\frac{p}{p^2+a^2}-\frac{p}{p^2+b^2}\Biggr)\mathrm{d}p.
  \end{align*}
  原函数的不定积分是
  \begin{equation*}
    \int\frac{p\,\mathrm{d}p}{p^2+a^2}=\frac12\ln(p^2+a^2).
  \end{equation*}
  因此
  \begin{equation*}
    \Biggl[\frac12\ln\frac{p^2+a^2}{p^2+b^2}\Biggr]_0^{\infty}
    =0-\frac12\ln\frac{a^2}{b^2}
    =\ln\frac{b}{a}.
  \end{equation*}
  上限：分子分母都像 $p^2$，对数趋于 $\ln 1=0$。
\end{frame}

\begin{frame}{例：简谐振子}
  质量为 $m$ 的质点在弹性系数为 $k$ 的弹簧牵引下做简谐运动，
  \begin{equation*}
    m\frac{\mathrm{d}^2 X}{\mathrm{d}t^2}+k X=0,
  \end{equation*}
  初始条件取 $X(0)=X_0$，$X'(0)=0$。
  对方程作拉普拉斯变换，用 $x(p)$ 表示 $\mathcal{L}\{X\}$。
  二阶导数定理
  \begin{equation*}
    \mathcal{L}\{X''\}=p^2 x(p)-p X(0)-X'(0)
  \end{equation*}
  代入后，
  \begin{equation*}
    m\bigl(p^2 x-p X_0\bigr)+k x=0.
  \end{equation*}
\end{frame}

\begin{frame}{解出原函数}
  令 $\omega_0^2=k/m$，
  \begin{equation*}
    x(p)\bigl(mp^2+k\bigr)=mp X_0,
    \qquad
    x(p)=X_0\frac{p}{p^2+\omega_0^2}.
  \end{equation*}
  这正是 $\cos(\omega_0 t)$ 的像函数，所以
  \begin{equation*}
    X(t)=X_0\cos(\omega_0 t).
  \end{equation*}
  \vspace{0.4em}

  若 $X'(0)=V_0$，导数定理多一项 $-m V_0$，
  \begin{equation*}
    x(p)=X_0\frac{p}{p^2+\omega_0^2}
        +\frac{V_0}{p^2+\omega_0^2}.
  \end{equation*}
  第二项是 $\sin(\omega_0 t)/\omega_0$ 的像，
  \begin{equation*}
    X(t)=X_0\cos(\omega_0 t)
         +\frac{V_0}{\omega_0}\sin(\omega_0 t).
  \end{equation*}
\end{frame}

\begin{frame}{例：积分方程}
  \begin{equation*}
    y(t)=1+\int_0^t y(\tau)\sin(t-\tau)\,\mathrm{d}\tau.
  \end{equation*}
  积分是 $y$ 与 $\sin t$ 的卷积。两边作拉普拉斯变换，
  \begin{equation*}
    Y(p)=\frac{1}{p}+Y(p)\frac{1}{p^2+1}.
  \end{equation*}
  解出
  \begin{equation*}
    Y(p)=\frac{p^2+1}{p^3}=\frac{1}{p}+\frac{1}{p^3},
  \end{equation*}
  所以 $y(t)=1+t^2/2$。
\end{frame}

\begin{frame}{例：$\lambda$ 的三种情形}
  $x''+\lambda x=0$，$x(0)=\phi$，$\dot x(0)=\psi$。导数定理给出
  \begin{equation*}
    X(p)=\phi\frac{p}{p^2+\lambda}+\psi\frac{1}{p^2+\lambda}.
  \end{equation*}
  \begin{itemize}
    \item $\lambda=a^2>0$：
      $x=\phi\cos(at)+(\psi/a)\sin(at)$。
    \item $\lambda=0$：
      $X=\phi/p+\psi/p^2$，故 $x=\phi+\psi t$。
    \item $\lambda=-b^2<0$：
      $x=\phi\cosh(bt)+(\psi/b)\sinh(bt)$。
  \end{itemize}
  三种情形都不必另配齐次通解：初值已在运算规则里用完。
\end{frame}

\begin{frame}{例：非齐次方程}
  $y''+y=2$，$y(0)=0$，$y'(0)=1$。变换后
  \begin{equation*}
    (p^2+1)Y(p)-1=\frac{2}{p},
    \qquad
    Y(p)=\frac{1}{p^2+1}+\frac{2}{p(p^2+1)}.
  \end{equation*}
  第二项裂成
  \begin{equation*}
    \frac{2}{p(p^2+1)}=\frac{2}{p}-\frac{2p}{p^2+1}.
  \end{equation*}
  逐项反演，
  \begin{equation*}
    y(t)=\sin t+2-2\cos t.
  \end{equation*}
  也可用留数：极点 $p=0,\pm\mathrm{i}$，对 $Y(p)e^{pt}$ 求和，结果相同。
\end{frame}

\begin{frame}{例：矩形脉冲驱动力}
  $y''+4y=1-u(t-1)$，$y(0)=1$，$y'(0)=0$。
  $\mathcal{L}\{u(t-1)\}=e^{-p}/p$，故
  \begin{equation*}
    (p^2+4)Y-p=\frac{1-e^{-p}}{p}.
  \end{equation*}
  裂项
  $\displaystyle\frac{1}{p(p^2+4)}=\frac14\Bigl(\frac{1}{p}-\frac{p}{p^2+4}\Bigr)$，
  反演并用延迟定理，
  \begin{equation*}
    y(t)=\cos 2t+\frac14\bigl(1-\cos 2t\bigr)
      -\frac14 u(t-1)\bigl[1-\cos 2(t-1)\bigr].
  \end{equation*}
  $t=1$ 撤去外力后，$y$ 与 $y'$ 仍连续，$y''$ 跳一个单位。
\end{frame}

\begin{frame}{例：阻尼受迫振子}
  $x''+2\gamma x'+\omega_0^2 x=f(t)$，$x(0)=\phi$，$\dot x(0)=\psi$。像函数是
  \begin{equation*}
    X(p)=\frac{(p+2\gamma)\phi+\psi+F(p)}{p^2+2\gamma p+\omega_0^2}.
  \end{equation*}
  令 $R=\omega_0^2-\gamma^2$。自由运动由 $R$ 的符号决定，强迫项是对应核与 $f$ 的卷积。
  \begin{itemize}
    \item $R=a^2>0$：核是 $e^{-\gamma t}\sin(at)$，自由运动为
      $e^{-\gamma t}\bigl[\phi\cos(at)+(\gamma\phi+\psi)\sin(at)/a\bigr]$。
    \item $R=0$：自由运动为 $e^{-\gamma t}\bigl[\phi+(\gamma\phi+\psi)t\bigr]$。
    \item $R=-b^2<0$：把 $\sin$、$\cos$ 换成 $\sinh$、$\cosh$。
  \end{itemize}
\end{frame}

\begin{frame}{本讲小结}
  \begin{itemize}
    \item Bromwich 积分 $=$ 左半平面留数之和；简单极点与高阶极点公式。
    \item 例：$\omega/(p^2+\omega^2)\to\sin\omega t$；
      $1/[p(p+a)]\to(1-e^{-at})/a$；$1/(p+\alpha)^2\to t e^{-\alpha t}$。
    \item 裂项与支点：$1/\sqrt{p}\to 1/\sqrt{\pi t}$。
    \item 应用：级数和、Dirichlet 积分、初值问题、脉冲驱动、阻尼振子。
  \end{itemize}
\end{frame}
