\begin{frame}{从级数到积分}
  上一讲的对象是周期函数，频率是 $\omega_n=n\pi/\ell$，一份一份地增加。
  \vspace{0.4em}

  非周期函数可以看成周期 $2\ell$ 趋于无穷。频率间隔 $\Delta\omega=\pi/\ell$ 趋于零，求和变成积分。
  本讲固定一种归一化，把性质和三个例子都按同一种约定来算。
\end{frame}

\begin{frame}{本讲提纲}
  \begin{enumerate}
    \item 由复傅里叶级数取 $\ell\to\infty$，写出傅里叶变换对；
    \item $\delta$ 函数的极限、挑选性、复合，以及 $e^{-\alpha|x|}$、狭缝和高斯；
    \item 导数、延迟和卷积；
    \item 二维矩孔的衍射，作为高维变换的例子。
  \end{enumerate}
\end{frame}

\section{从级数取极限}

\begin{frame}{频率越来越密}
  周期 $2\ell$ 的复级数是
  \begin{equation*}
    f(x)=\sum_{n=-\infty}^{\infty} c_n e^{\mathrm{i}\omega_n x},
    \qquad
    \omega_n=\frac{n\pi}{\ell},
  \end{equation*}
  \begin{equation*}
    c_n=\frac{1}{2\ell}\int_{-\ell}^{\ell} f(x)\,e^{-\mathrm{i}\omega_n x}\,\mathrm{d}x.
  \end{equation*}
  相邻频率的间隔是 $\Delta\omega=\pi/\ell$，所以 $1/(2\ell)=\Delta\omega/(2\pi)$。代入后
  \begin{equation*}
    f(x)=\sum_n \frac{\Delta\omega}{2\pi}
      \Bigl(\int_{-\ell}^{\ell} f(t)\,e^{-\mathrm{i}\omega_n t}\,\mathrm{d}t\Bigr)
      e^{\mathrm{i}\omega_n x}.
  \end{equation*}
\end{frame}

\begin{frame}{求和变成积分}
  $\ell\to\infty$ 时 $\Delta\omega\to 0$，求和变成对 $\omega$ 的积分：
  \begin{align*}
    F_{\mathrm{full}}(\omega)
      &=\int_{-\infty}^{\infty} f(x)\,e^{-\mathrm{i}\omega x}\,\mathrm{d}x,\\
    f(x)
      &=\frac{1}{2\pi}\int_{-\infty}^{\infty}
        F_{\mathrm{full}}(\omega)\,e^{\mathrm{i}\omega x}\,\mathrm{d}\omega.
  \end{align*}
  $2\pi$ 全部放在还原原函数的那一次积分上。把常数拆成两半，就得到本讲使用的对称形式。
\end{frame}

\begin{frame}{对称形式}
  \begin{equation*}
    \boxed{
    \begin{aligned}
      F(\omega)&=\frac{1}{\sqrt{2\pi}}
        \int_{-\infty}^{\infty} f(x)\,e^{-\mathrm{i}\omega x}\,\mathrm{d}x,\\
      f(x)&=\frac{1}{\sqrt{2\pi}}
        \int_{-\infty}^{\infty} F(\omega)\,e^{\mathrm{i}\omega x}\,\mathrm{d}\omega.
    \end{aligned}}
  \end{equation*}
  记 $F=\mathcal{F}[f]$，$f=\mathcal{F}^{-1}[F]$。$f$ 是原函数，$F$ 是像函数。
  \vspace{0.3em}

  两种写法差的只是常数放在哪一边。导数定理里的 $\mathrm{i}\omega$ 由 $e^{-\mathrm{i}\omega x}$ 决定，不由常数决定。
\end{frame}

\begin{frame}{什么时候积分存在}
  傅里叶积分定理：若
  \begin{enumerate}
    \item $f$ 在任一有限区间上满足狄利克雷条件；
    \item $\displaystyle\int_{-\infty}^{\infty}|f(x)|\,\mathrm{d}x$ 收敛，
  \end{enumerate}
  则 $f$ 可以写成傅里叶积分。间断点上，积分值仍是左右极限的平均。
  \vspace{0.4em}

  $e^{\alpha x}$（$\alpha>0$）在正半轴上不衰减，不满足第二条。第三讲用衰减因子 $e^{-\sigma t}$ 把它纳进来。
\end{frame}

\begin{frame}{正余弦形式}
  实的原函数也可以写成
  \begin{equation*}
    f(x)=\int_0^{\infty}
      \bigl[A(\omega)\cos(\omega x)+B(\omega)\sin(\omega x)\bigr]\,\mathrm{d}\omega,
  \end{equation*}
  \begin{align*}
    A(\omega)&=\frac{1}{\pi}\int_{-\infty}^{\infty} f(x)\cos(\omega x)\,\mathrm{d}x,\\
    B(\omega)&=\frac{1}{\pi}\int_{-\infty}^{\infty} f(x)\sin(\omega x)\,\mathrm{d}x.
  \end{align*}
  合成振幅和相位
  \begin{equation*}
    C=\sqrt{A^2+B^2},
    \qquad
    \varphi=\arctan(B/A),
  \end{equation*}
  则 $f(x)=\int_0^{\infty} C(\omega)\cos[\omega x-\varphi(\omega)]\,\mathrm{d}\omega$。
  $C$ 是振幅谱，$\varphi$ 是相位谱。偶函数只有余弦积分，奇函数只有正弦积分。
\end{frame}

\section{三个像函数}

\begin{frame}{$\delta$ 函数}
  质量 $m$ 均匀铺在长为 $l$ 的线段上，线密度是 $m/l$。令 $l\to 0$，密度在原点变成无穷，其余点是零，但积分仍是 $m$。
  \vspace{0.3em}

  把总质量取成 $1$，就得到
  \begin{equation*}
    \delta(x)=0\quad(x\neq 0),
    \qquad
    \int_a^b \delta(x)\,\mathrm{d}x=
    \begin{cases}
      1, & a<0<b,\\
      0, & \text{其他}.
    \end{cases}
  \end{equation*}
  它是广义函数，不是普通函数在每一点的值。
\end{frame}

\begin{frame}{挑选性}
  若 $f$ 在 $a$ 连续，
  \begin{equation*}
    \int_{-\infty}^{\infty} f(x)\,\delta(x-a)\,\mathrm{d}x=f(a).
  \end{equation*}
  $\delta$ 是偶函数，$\delta(-x)=\delta(x)$；它的导数是奇函数。缩放时
  \begin{equation*}
    \delta(ax)=\frac{\delta(x)}{|a|}\qquad(a\neq 0).
  \end{equation*}
\end{frame}

\begin{frame}{它是哪几种极限}
  下面三个函数的积分都是 $1$，参数趋于极端时都挤到原点：
  \begin{align*}
    \delta(x)
      &=\lim_{l\to 0}
        \begin{cases}
          1/l, & |x|\le l/2,\\
          0, & \text{其他},
        \end{cases}\\
    \delta(x)
      &=\lim_{K\to\infty}\frac{\sin(Kx)}{\pi x},\\
    \delta(x)
      &=\lim_{\varepsilon\to 0^+}
        \frac{1}{\pi}\frac{\varepsilon}{\varepsilon^2+x^2}.
  \end{align*}
  方脉冲越来越窄，主瓣越来越窄，洛伦兹峰越来越尖。用到的是这些函数的积分，不是它们在原点的函数值。
\end{frame}

\begin{frame}{导数也有挑选性}
  对光滑的 $f$，分部积分把 $\delta'$ 上的导数挪到 $f$ 上，边界项是零：
  \begin{equation*}
    \int_{-\infty}^{\infty} f(x)\,\delta'(x-a)\,\mathrm{d}x=-f'(a).
  \end{equation*}
  所以 $\delta'$ 挑选的是导数，并且带一个负号。$\delta'$ 是奇函数，和这个负号是同一件事。
\end{frame}

\begin{frame}{自变量是另一个函数}
  若 $\varphi$ 的实零点 $x_k$ 都是单零点，
  \begin{equation*}
    \boxed{\delta\bigl(\varphi(x)\bigr)
      =\sum_k\frac{\delta(x-x_k)}{|\varphi'(x_k)|}.}
  \end{equation*}
  $\delta(ax)=\delta(x)/|a|$ 是只有一个零点的特例。两个零点时，$a\neq 0$，
  \begin{equation*}
    \delta(x^2-a^2)
    =\frac{\delta(x-a)+\delta(x+a)}{2|a|}.
  \end{equation*}
\end{frame}

\begin{frame}{例：三个点电荷}
  电荷 $-e$、$2e$、$-e$ 分别放在 $x=-4$、$0$、$\pi$。线密度是
  \begin{equation*}
    \rho(x)=-e\,\delta(x+4)+2e\,\delta(x)-e\,\delta(x-\pi).
  \end{equation*}
  总电荷是系数之和，等于 $0$。区间 $(-3,3)$ 只含 $x=0$ 这一点，所以这段上的电荷是 $2e$。
\end{frame}

\begin{frame}{例：$\delta(x^2-4)$}
  零点是 $x=\pm 2$，而 $|\varphi'(x)|=|2x|=4$。因此
  \begin{align*}
    \int_{-\infty}^{\infty}(x^2+1)\,\delta(x^2-4)\,\mathrm{d}x
    &=\frac{2^2+1}{4}+\frac{(-2)^2+1}{4}
    =\frac{5}{2}.
  \end{align*}
  每个零点贡献 $f(x_k)/|\varphi'(x_k)|$，两处加起来。
\end{frame}

\begin{frame}{$\delta$ 的傅里叶积分}
  对称约定下 $\mathcal{F}[\delta]=1/\sqrt{2\pi}$。代回逆变换，
  \begin{equation*}
    \delta(x)
    =\frac{1}{2\pi}\int_{-\infty}^{\infty}e^{\mathrm{i}\omega x}\,\mathrm{d}\omega.
  \end{equation*}
  这个积分不是普通收敛的。先积到有限频率 $K$，得到 $\sin(Kx)/(\pi x)$，再令 $K\to\infty$。平坦的频谱叠加之后，只在 $x=0$ 处相长。
\end{frame}

\begin{frame}{例：$e^{-\alpha|x|}$}
  $\alpha>0$。按对称形式，并使用 $e^{-\mathrm{i}\omega x}$，
  \begin{align*}
    \int_0^{\infty} e^{-\alpha x}e^{-\mathrm{i}\omega x}\,\mathrm{d}x
      &=\frac{1}{\alpha+\mathrm{i}\omega},\\
    \int_{-\infty}^{0} e^{\alpha x}e^{-\mathrm{i}\omega x}\,\mathrm{d}x
      &=\frac{1}{\alpha-\mathrm{i}\omega}.
  \end{align*}
  两段相加是 $2\alpha/(\alpha^2+\omega^2)$。因此
  \begin{equation*}
    F(\omega)=\sqrt{\frac{1}{2\pi}}\,\frac{2\alpha}{\alpha^2+\omega^2}.
  \end{equation*}
  $f$ 是偶函数，像函数是实的。
\end{frame}

\begin{frame}{越窄的脉冲，频谱越宽}
  $\alpha$ 越大，$e^{-\alpha|x|}$ 越集中在 $x=0$ 附近，而 $2\alpha/(\alpha^2+\omega^2)$ 在 $\omega$ 方向越平坦。
  \vspace{0.4em}

  极端情形是 $\delta(x)$：它在一点以外全是零。代入定义，
  \begin{equation*}
    \mathcal{F}[\delta]
    =\frac{1}{\sqrt{2\pi}}\int_{-\infty}^{\infty}
      \delta(x)\,e^{-\mathrm{i}\omega x}\,\mathrm{d}x
    =\frac{1}{\sqrt{2\pi}}.
  \end{equation*}
  原函数无限局域，像函数是常数，无限铺开。
\end{frame}

\begin{frame}{例：高斯函数}
  $\alpha>0$，$f(x)=e^{-\alpha x^2}$。配平方：
  \begin{equation*}
    -\alpha x^2-\mathrm{i}\omega x
    =-\alpha\Bigl(x+\frac{\mathrm{i}\omega}{2\alpha}\Bigr)^2
      -\frac{\omega^2}{4\alpha}.
  \end{equation*}
  积分平移到复平面后仍等于实轴上的高斯积分 $\sqrt{\pi/\alpha}$，所以
  \begin{equation*}
    \boxed{F(\omega)=\frac{1}{\sqrt{2\alpha}}\,
      e^{-\omega^2/(4\alpha)}.}
  \end{equation*}
  高斯函数的像还是高斯函数。$\alpha$ 越大，原函数越窄，像函数的指数衰减越慢，又是同一条宽窄对偶。
\end{frame}

\begin{frame}{例：一维狭缝}
  设 $f(x)=1$（$|x|\le a$），其余为零，$a>0$。
  \begin{equation*}
    F(\omega)
    =\frac{1}{\sqrt{2\pi}}\int_{-a}^{a}e^{-\mathrm{i}\omega x}\,\mathrm{d}x
    =\sqrt{\frac{2}{\pi}}\,a\,\frac{\sin(a\omega)}{a\omega}.
  \end{equation*}
  $\omega=0$ 时商式的极限是 $\sqrt{2/\pi}\,a$，也就是 $(1/\sqrt{2\pi})\int f$。
  缝越宽，中央峰越窄。高维那一节的矩孔就是两个这样的因子相乘。
\end{frame}

\section{基本性质}

\begin{frame}{导数定理}
  设边界项在无穷远处消失。分部积分：
  \begin{equation*}
    \int_{-\infty}^{\infty} f'(x)\,e^{-\mathrm{i}\omega x}\,\mathrm{d}x
    =\mathrm{i}\omega\int_{-\infty}^{\infty} f(x)\,e^{-\mathrm{i}\omega x}\,\mathrm{d}x.
  \end{equation*}
  两边同除 $\sqrt{2\pi}$，
  \begin{equation*}
    \boxed{\mathcal{F}[f']=\mathrm{i}\omega\,F(\omega).}
  \end{equation*}
  求导变成乘 $\mathrm{i}\omega$。这是用傅里叶变换解微分方程的原因，也是下一讲拉普拉斯变换里 $p\bar f(p)-f(0)$ 的前身。
\end{frame}

\begin{frame}{积分、相似、延迟、位移}
  在同一对称约定下：
  \begin{align*}
    \mathcal{F}\Bigl[\int^{x} f\Bigr]
      &=\frac{1}{\mathrm{i}\omega}F(\omega),\\
    \mathcal{F}[f(ax)]
      &=\frac{1}{|a|}F\Bigl(\frac{\omega}{a}\Bigr)
      \quad(a\neq 0),\\
    \mathcal{F}[f(x-x_0)]
      &=e^{-\mathrm{i}\omega x_0}F(\omega),\\
    \mathcal{F}[e^{\mathrm{i}\omega_0 x}f(x)]
      &=F(\omega-\omega_0).
  \end{align*}
  相似定理就是宽窄对偶的一般说法：把原函数压窄 $a$ 倍，像函数变宽 $a$ 倍，并乘上 $1/|a|$。
\end{frame}

\begin{frame}{延迟定理从哪来}
  \begin{align*}
    \mathcal{F}[f(x-x_0)]
    &=\frac{1}{\sqrt{2\pi}}\int_{-\infty}^{\infty}
      f(x-x_0)\,e^{-\mathrm{i}\omega x}\,\mathrm{d}x.
  \end{align*}
  令 $u=x-x_0$，指数多出一个因子 $e^{-\mathrm{i}\omega x_0}$，积分变回 $F(\omega)$。
  \vspace{0.4em}

  位移定理是同一计算的对偶：原函数乘 $e^{\mathrm{i}\omega_0 x}$，像函数的频率原点挪到 $\omega_0$。
\end{frame}

\begin{frame}{卷积定理}
  标准卷积是
  \begin{equation*}
    (f_1*f_2)(x)
    =\int_{-\infty}^{\infty} f_1(\alpha)\,f_2(x-\alpha)\,\mathrm{d}\alpha.
  \end{equation*}
  代入对称形式的定义，交换积分次序，得到两个完整的傅里叶变换，并剩下一个 $\sqrt{2\pi}$：
  \begin{equation*}
    \boxed{\mathcal{F}[f_1*f_2]
      =\sqrt{2\pi}\,F_1(\omega)\,F_2(\omega).}
  \end{equation*}
  常数来自归一化，不是卷积本身。若把 $2\pi$ 全部放在逆变换上，这个因子会换成别的常数；乘积结构不变。
\end{frame}

\begin{frame}{和 $\delta$ 卷积就是平移}
  \begin{equation*}
    \int_{-\infty}^{\infty} f(x-\alpha)\,\delta(\alpha-x_0)\,\mathrm{d}\alpha
    =f(x-x_0).
  \end{equation*}
  卷积定理再写一遍：$\mathcal{F}[\delta(\cdot-x_0)]=e^{-\mathrm{i}\omega x_0}/\sqrt{2\pi}$，于是
  \begin{equation*}
    \sqrt{2\pi}\,F(\omega)\cdot\frac{e^{-\mathrm{i}\omega x_0}}{\sqrt{2\pi}}
    =e^{-\mathrm{i}\omega x_0}F(\omega).
  \end{equation*}
  这就是延迟定理。平移可以看成和一只移动后的 $\delta$ 做卷积。
\end{frame}

\section{高维}

\begin{frame}{二维}
  两个自变量时，对称约定是每一侧各有一个 $1/(2\pi)$：
  \begin{align*}
    F(k_1,k_2)
      &=\frac{1}{2\pi}\iint
        f(x,y)\,e^{-\mathrm{i}(k_1 x+k_2 y)}\,\mathrm{d}x\,\mathrm{d}y,\\
    f(x,y)
      &=\frac{1}{2\pi}\iint
        F(k_1,k_2)\,e^{\mathrm{i}(k_1 x+k_2 y)}\,\mathrm{d}k_1\,\mathrm{d}k_2.
  \end{align*}
  一维的 $\sqrt{2\pi}$ 换成 $(2\pi)^{d/2}$。三维就是 $(2\pi)^{3/2}$。
\end{frame}

\begin{frame}{例：矩孔}
  矩形孔内振幅为常数 $A$，孔外为零：
  \begin{equation*}
    f(x,y)=
    \begin{cases}
      A, & |x|\leq X,\ |y|\leq Y,\\
      0, & \text{其他}.
    \end{cases}
  \end{equation*}
  二重积分分开：
  \begin{equation*}
    \int_{-X}^{X} e^{-\mathrm{i}k_1 x}\,\mathrm{d}x
    =\frac{2\sin(k_1 X)}{k_1}.
  \end{equation*}
  $y$ 方向同样。因此
  \begin{equation*}
    F(k_1,k_2)
    =\frac{2AXY}{\pi}
      \frac{\sin(k_1 X)}{k_1 X}
      \frac{\sin(k_2 Y)}{k_2 Y}.
  \end{equation*}
\end{frame}

\begin{frame}{矩孔的衍射图样}
  $\sin(kX)/(kX)$ 在 $k=0$ 处为 $1$，并在 $kX=n\pi$（$n\neq 0$）处为零。
  \vspace{0.4em}

  所以衍射强度的主峰在正前方，旁边有一串暗纹。孔越窄（$X$ 越小），中央亮斑越宽。这仍是“空域越窄，频域越宽”，只是频率换成了波矢 $(k_1,k_2)$。
\end{frame}

\begin{frame}{三维}
  $\mathbf{k}=(k_1,k_2,k_3)$，$\mathbf{r}=(x,y,z)$。对称形式是
  \begin{align*}
    F(\mathbf{k})
      &=\frac{1}{(2\pi)^{3/2}}
        \iiint f(\mathbf{r})\,
        e^{-\mathrm{i}\mathbf{k}\cdot\mathbf{r}}\,\mathrm{d}^3\mathbf{r},\\
    f(\mathbf{r})
      &=\frac{1}{(2\pi)^{3/2}}
        \iiint F(\mathbf{k})\,
        e^{\mathrm{i}\mathbf{k}\cdot\mathbf{r}}\,\mathrm{d}^3\mathbf{k}.
  \end{align*}
  计算步骤和矩孔相同：指数可分离时，积分拆成三个一维积分。
\end{frame}

\begin{frame}{三维 $\delta$ 函数}
  直角坐标里 $\delta(\mathbf{r})=\delta(x)\delta(y)\delta(z)$，三重积分等于 $1$。
  \vspace{0.35em}

  场点不在原点、也不在球坐标的极轴上时，分母是体积元的雅可比：
  \begin{align*}
    \delta(\mathbf{r}-\mathbf{r}')
      &=\frac{\delta(\rho-\rho')\,\delta(\phi-\phi')\,\delta(z-z')}{\rho},\\
    \delta(\mathbf{r}-\mathbf{r}')
      &=\frac{\delta(r-r')\,\delta(\theta-\theta')\,\delta(\phi-\phi')}{r^2\sin\theta}.
  \end{align*}
  柱坐标除以 $\rho$，球坐标除以 $r^2\sin\theta$。这样对 $\delta$ 积分才和体积元配成 $1$。
\end{frame}

\begin{frame}{本讲小结}
  \begin{itemize}
    \item 周期趋于无穷后，$c_n$ 变成 $F(\omega)$。本讲把 $2\pi$ 对称地放在正、逆变换两侧。
    \item $\delta$ 由窄脉冲取极限得到。复合函数按零点拆开；和 $\delta$ 卷积就是平移。
    \item $e^{-\alpha|x|}$、狭缝和高斯函数说明：原函数越集中，像函数越铺开。
    \item 求导变成乘 $\mathrm{i}\omega$，平移变成乘相位，卷积变成乘积。
  \end{itemize}
  下一讲处理不绝对可积、但只在 $t>0$ 有定义的函数。
\end{frame}
