\section{傅里叶变换}

\begin{frame}{从周期到非周期}
  前面讨论了周期性体系的傅里叶级数展开。现在研究定义在 $-\infty<x<\infty$ 上的实函数。依旧对该函数进行傅里叶级数展开，该级数必须理解为周期 $2\ell$ 扩展至 $\infty$ 的极限情况。
  \vspace{0.4em}

  三角函数族的变量为 $n\pi x/\ell$。引入
  \begin{equation*}
    \omega_n=\frac{n\pi}{\ell}=n\omega_0,
    \qquad
    \omega_0=\frac{\pi}{\ell},
  \end{equation*}
  $\omega_0$ 称为\textbf{基础频率}。相邻频率之差
  \begin{equation*}
    \Delta\omega_n=\omega_n-\omega_{n-1}=\omega_0=\frac{\pi}{\ell}.
  \end{equation*}
\end{frame}

\begin{frame}{复级数与系数}
  \begin{equation*}
    f(x)=\sum_{n=-\infty}^{\infty} c_n\,e^{\mathrm{i}\omega_n x},
    \qquad
    c_n=\frac{1}{2\ell}\int_{-\ell}^{\ell}
      f(x)\,e^{-\mathrm{i}\omega_n x}\,\mathrm{d}x.
  \end{equation*}
  由 $\Delta\omega=\pi/\ell$ 得 $2\ell=2\pi/\Delta\omega$，所以
  \begin{equation*}
    \frac{1}{2\ell}=\frac{\Delta\omega}{2\pi}.
  \end{equation*}
  代回级数，
  \begin{equation*}
    f(x)=\sum_{n}
      \Biggl[
        \frac{1}{2\pi}\int_{-\ell}^{\ell}
          f(t)\,e^{-\mathrm{i}\omega_n t}\,\mathrm{d}t
      \Biggr]
      e^{\mathrm{i}\omega_n x}\,\Delta\omega.
  \end{equation*}
\end{frame}

\begin{frame}{求和变成积分}
  取 $\ell\to\infty$，则 $\Delta\omega\to 0$，
  \begin{equation*}
    \sum_n(\,\cdot\,)\,\Delta\omega
    \longrightarrow\int_{-\infty}^{\infty}(\,\cdot\,)\,\mathrm{d}\omega.
  \end{equation*}
  同一极限也可以写成 $\sum_n\to(\ell/\pi)\int\mathrm{d}\omega$，因为 $1/\Delta\omega=\ell/\pi$。
  记
  \begin{equation*}
    F(\omega)
    \equiv\lim_{\ell\to\infty}\frac{2\ell\,c_n}{2\pi}
    =\frac{1}{2\pi}\int_{-\infty}^{\infty}
      e^{-\mathrm{i}\omega x}f(x)\,\mathrm{d}x,
  \end{equation*}
  则
  \begin{equation*}
    f(x)=\int_{-\infty}^{\infty}
      F(\omega)\,e^{\mathrm{i}\omega x}\,\mathrm{d}\omega.
  \end{equation*}
  上式的积分称为\textbf{傅里叶积分}。这里积分前的系数并不要紧，更关键的是积分关于 $\omega$ 的形式。
\end{frame}

\begin{frame}{傅里叶积分定理}
  \textbf{傅里叶积分定理}：若 $f(x)$ 在 $(-\infty,\infty)$ 上满足：
  \begin{enumerate}
    \item 在任一有限区间上满足狄里希利条件；
    \item 在 $(-\infty,\infty)$ 上绝对可积，即 $\displaystyle\int_{-\infty}^{\infty}|f(x)|\,\mathrm{d}x$ 收敛，
  \end{enumerate}
  则 $f(x)$ 可表示成傅里叶积分，且在间断点处
  \begin{equation*}
    \text{傅里叶积分值}
    =\lim_{\epsilon\to 0}\frac12
      \bigl[f(x+\epsilon)+f(x-\epsilon)\bigr].
  \end{equation*}
\end{frame}

\begin{frame}{对称形式}
  复数形式的傅里叶积分可以写成对称的形式
  \begin{equation*}
    \begin{aligned}
      f(x)&=\frac{1}{\sqrt{2\pi}}\int_{-\infty}^{\infty}
        F(\omega)\,e^{\mathrm{i}\omega x}\,\mathrm{d}\omega,\\
      F(\omega)&=\frac{1}{\sqrt{2\pi}}\int_{-\infty}^{\infty}
        f(x)\bigl[e^{\mathrm{i}\omega x}\bigr]^*\,\mathrm{d}x.
    \end{aligned}
  \end{equation*}
  并常用符号简写为
  \begin{equation*}
    F(\omega)=\mathcal{F}[f(x)],
    \qquad
    f(x)=\mathcal{F}^{-1}[F(\omega)].
  \end{equation*}
  $f(x)$ 和 $F(\omega)$ 分别称为傅里叶变换的\textbf{原函数}和\textbf{像函数}。
  本讲下面的例子都用这个对称形式。
\end{frame}

\begin{frame}{正余弦形式：$c_n$ 与 $a_n$、$b_n$}
  上一讲已有（$n>0$）
  \begin{equation*}
    c_n=\tfrac12(a_n-\mathrm{i}b_n),
    \qquad
    c_{-n}=\tfrac12(a_n+\mathrm{i}b_n),
  \end{equation*}
  亦即
  \begin{equation*}
    a_n=c_n+c_{-n},
    \qquad
    b_n=\mathrm{i}(c_n-c_{-n}).
  \end{equation*}
  把 $\pm n$ 两项合并，
  \begin{equation*}
    c_n e^{\mathrm{i}\omega_n x}+c_{-n}e^{-\mathrm{i}\omega_n x}
    =a_n\cos(\omega_n x)+b_n\sin(\omega_n x).
  \end{equation*}
  又 $F(\omega)=\displaystyle\lim\frac{2\ell\,c_n}{2\pi}=\lim\frac{c_n}{\Delta\omega}$，
  相应定义
  \begin{equation*}
    A(\omega)=\lim\frac{a_n}{\Delta\omega}=F(\omega)+F(-\omega),
    \quad
    B(\omega)=\lim\frac{b_n}{\Delta\omega}
      =\mathrm{i}\bigl[F(\omega)-F(-\omega)\bigr].
  \end{equation*}
  等价地，$F(\omega)=\tfrac12\bigl[A(\omega)-\mathrm{i}B(\omega)\bigr]$。
\end{frame}

\begin{frame}{正余弦形式}
  由 $a_n=(1/\ell)\int_{-\ell}^{\ell}f\cos$，$1/\ell=\Delta\omega/\pi$，得
  \begin{equation*}
    A(\omega)=\frac{1}{\pi}\int_{-\infty}^{\infty} f(t)\cos(\omega t)\,\mathrm{d}t,
    \quad
    B(\omega)=\frac{1}{\pi}\int_{-\infty}^{\infty} f(t)\sin(\omega t)\,\mathrm{d}t.
  \end{equation*}
  级数中 $\pm n$ 合并后再取 $\ell\to\infty$，
  \begin{equation*}
    \sum_{n=1}^{\infty}
      \bigl[a_n\cos(\omega_n x)+b_n\sin(\omega_n x)\bigr]
    \longrightarrow
    \int_0^{\infty}
      \bigl[A(\omega)\cos(\omega x)+B(\omega)\sin(\omega x)\bigr]
      \,\mathrm{d}\omega,
  \end{equation*}
  即
  \begin{equation*}
    f(x)=\int_0^{\infty} A(\omega)\cos(\omega x)\,\mathrm{d}\omega
        +\int_0^{\infty} B(\omega)\sin(\omega x)\,\mathrm{d}\omega.
  \end{equation*}
  偶函数时 $B=0$（\textbf{傅里叶余弦积分}）；奇函数时 $A=0$（\textbf{傅里叶正弦积分}）.
\end{frame}

\begin{frame}{振幅谱与相位谱}
  上式还可以写成
  \begin{equation*}
    f(x)=\int_0^{\infty}
      C(\omega)\cos\bigl[\omega x-\varphi(\omega)\bigr]\,\mathrm{d}\omega,
  \end{equation*}
  其中
  \begin{align*}
    C(\omega)&=\sqrt{[A(\omega)]^2+[B(\omega)]^2},\\
    \varphi(\omega)&=\arctan\bigl[B(\omega)/A(\omega)\bigr].
  \end{align*}
  核对：$A\cos(\omega x)+B\sin(\omega x)
    =C\bigl[\cos\varphi\cos(\omega x)+\sin\varphi\sin(\omega x)\bigr]
    =C\cos(\omega x-\varphi)$，
  因为 $\cos\varphi=A/C$，$\sin\varphi=B/C$。
  $C(\omega)$ 称为 $f(x)$ 的\textbf{振幅谱}，$\varphi(\omega)$ 称为 $f(x)$ 的\textbf{相位谱}。
\end{frame}




\begin{frame}{例：$e^{-\alpha|x|}$}
  取 $f(x)=e^{-\alpha|x|}$，$\alpha>0$。对称形式下在原点处拆开积分，
  \begin{align*}
    F(\omega)
    &=\frac{1}{\sqrt{2\pi}}
      \Biggl(
        \int_{-\infty}^{0}
          e^{\alpha x}e^{-\mathrm{i}\omega x}\,\mathrm{d}x
        +\int_0^{\infty}
          e^{-\alpha x}e^{-\mathrm{i}\omega x}\,\mathrm{d}x
      \Biggr)
    \\
    &=\frac{1}{\sqrt{2\pi}}
      \Biggl(
        \int_{-\infty}^{0}
          e^{(\alpha-\mathrm{i}\omega)x}\,\mathrm{d}x
        +\int_0^{\infty}
          e^{-(\alpha+\mathrm{i}\omega)x}\,\mathrm{d}x
      \Biggr).
  \end{align*}
\end{frame}

\begin{frame}{例：$e^{-\alpha|x|}$（续）}
  两段都是指数积分：
  \begin{align*}
    \int_{-\infty}^{0} e^{(\alpha-\mathrm{i}\omega)x}\,\mathrm{d}x
    &=\Biggl[
      \frac{e^{(\alpha-\mathrm{i}\omega)x}}{\alpha-\mathrm{i}\omega}
    \Biggr]_{-\infty}^{0}
    =\frac{1}{\alpha-\mathrm{i}\omega},
    \\
    \int_0^{\infty} e^{-(\alpha+\mathrm{i}\omega)x}\,\mathrm{d}x
    &=\Biggl[
      -\frac{e^{-(\alpha+\mathrm{i}\omega)x}}{\alpha+\mathrm{i}\omega}
    \Biggr]_0^{\infty}
    =\frac{1}{\alpha+\mathrm{i}\omega}.
  \end{align*}
  （$\mathrm{Re}\,\alpha>0$ 保证上限／下限收敛。）相加，
  \begin{equation*}
    F(\omega)
    =\frac{1}{\sqrt{2\pi}}
      \Biggl(
        \frac{1}{\alpha-\mathrm{i}\omega}
        +\frac{1}{\alpha+\mathrm{i}\omega}
      \Biggr)
    =\sqrt{\frac{1}{2\pi}}\,\frac{2\alpha}{\alpha^2+\omega^2}.
  \end{equation*}
  $F$ 实且偶，与实偶的 $f$ 一致。
\end{frame}

\begin{frame}{高斯函数的傅里叶变换}
  $f(x)=e^{-\alpha x^2}$，$\alpha>0$。对称形式下
  \begin{equation*}
    F(\omega)
    =\frac{1}{\sqrt{2\pi}}\int_{-\infty}^{\infty}
      e^{-\alpha x^2}e^{-\mathrm{i}\omega x}\,\mathrm{d}x.
  \end{equation*}
  指数配平方：
  \begin{align*}
    -\alpha x^2-\mathrm{i}\omega x
    &=-\alpha\Biggl[
        x^2+\frac{\mathrm{i}\omega}{\alpha}x
      \Biggr] \\
    &=-\alpha\Biggl(
        x+\frac{\mathrm{i}\omega}{2\alpha}
      \Biggr)^2
      -\frac{\omega^2}{4\alpha}.
  \end{align*}
  因此
  \begin{equation*}
    F(\omega)
    =\frac{e^{-\omega^2/(4\alpha)}}{\sqrt{2\pi}}
     \int_{-\infty}^{\infty}
       e^{-\alpha\bigl(x+\mathrm{i}\omega/(2\alpha)\bigr)^2}
       \,\mathrm{d}x.
  \end{equation*}
\end{frame}

\begin{frame}{高斯函数}
  因此平移后的积分等于实轴上的高斯积分：
  \begin{equation*}
    \int_{-\infty}^{\infty}
      e^{-\alpha\bigl(x+\mathrm{i}\omega/(2\alpha)\bigr)^2}\,\mathrm{d}x
    =\int_{-\infty}^{\infty} e^{-\alpha t^2}\,\mathrm{d}t
    =\sqrt{\frac{\pi}{\alpha}}.
  \end{equation*}
  代回配平方的结果，
  \begin{equation*}
    F(\omega)
    =\frac{e^{-\omega^2/(4\alpha)}}{\sqrt{2\pi}}
      \sqrt{\frac{\pi}{\alpha}}
    =\frac{1}{\sqrt{2\alpha}}\,e^{-\omega^2/(4\alpha)}.
  \end{equation*}
  高斯函数的像函数还是高斯函数。原函数越窄，像函数越宽。
\end{frame}

\section{基本性质}

\begin{frame}{基本性质}
  假定 $\mathcal{F}[f]=F$。在对称形式下：
  \begin{align*}
    \mathcal{F}[f']&=\mathrm{i}\omega\,F(\omega),\\
    \mathcal{F}\Bigl[\int^{x} f\Bigr]
      &=\frac{1}{\mathrm{i}\omega}F(\omega),\\
    \mathcal{F}[f(ax)]
      &=\frac{1}{|a|}F\Bigl(\frac{\omega}{a}\Bigr),\\
    \mathcal{F}[f(x-x_0)]
      &=e^{-\mathrm{i}\omega x_0}F(\omega),\\
    \mathcal{F}[e^{\mathrm{i}\omega_0 x}f]
      &=F(\omega-\omega_0).
  \end{align*}
  \textbf{卷积定理}放在后面单独演算。$a>0$ 时\textbf{相似定理}的分母就是 $a$。
\end{frame}

\begin{frame}{导数定理}
  \textbf{导数定理}由分部积分得到。
  \begin{align*}
    &\int_{-\infty}^{\infty} f'(x)\,e^{-\mathrm{i}\omega x}\,\mathrm{d}x \\
    &\quad=\Bigl[f\,e^{-\mathrm{i}\omega x}\Bigr]_{-\infty}^{\infty}
      +\mathrm{i}\omega\int_{-\infty}^{\infty}
        f\,e^{-\mathrm{i}\omega x}\,\mathrm{d}x.
  \end{align*}
  绝对可积且在无穷远处消失时，边界项为零。两边除以 $\sqrt{2\pi}$，
  \begin{equation*}
    \mathcal{F}[f']=\mathrm{i}\omega\,F(\omega).
  \end{equation*}
  \textbf{积分定理}是它的逆运算：记 $g(x)=\int^{x} f$，则 $g'=f$，
  $\mathrm{i}\omega\,\mathcal{F}[g]=F$，所以 $\mathcal{F}[g]=F/(\mathrm{i}\omega)$。
\end{frame}

\begin{frame}{相似、延迟、位移}
  $a>0$ 时令 $u=ax$，$\mathrm{d}x=\mathrm{d}u/a$，
  \begin{equation*}
    \mathcal{F}[f(ax)]
    =\frac{1}{\sqrt{2\pi}}\int f(u)\,
      e^{-\mathrm{i}(\omega/a)u}\,\frac{\mathrm{d}u}{a}
    =\frac{1}{a}F\Bigl(\frac{\omega}{a}\Bigr).
  \end{equation*}
  $a<0$ 时上下限对调，分母换成 $|a|$。
  \vspace{0.3em}

  \textbf{延迟定理}：令 $u=x-x_0$，指数多出 $e^{-\mathrm{i}\omega x_0}$。
  \textbf{位移定理}：
  \begin{equation*}
    \frac{1}{\sqrt{2\pi}}\int
      e^{\mathrm{i}\omega_0 x}f(x)\,e^{-\mathrm{i}\omega x}\,\mathrm{d}x
    =F(\omega-\omega_0).
  \end{equation*}
\end{frame}

\begin{frame}{卷积定理}
  \begin{equation*}
    (f_1*f_2)(x)
    =\int_{-\infty}^{\infty}
      f_1(\alpha)\,f_2(x-\alpha)\,\mathrm{d}\alpha
  \end{equation*}
  称为 $f_1$ 与 $f_2$ 的\textbf{卷积}。代入对称形式，
  \begin{align*}
    &\mathcal{F}[f_1*f_2]
    =\frac{1}{\sqrt{2\pi}}\int\mathrm{d}x\,e^{-\mathrm{i}\omega x}
      \int\mathrm{d}\alpha\,f_1(\alpha)\,f_2(x-\alpha).
  \end{align*}
  令 $u=x-\alpha$，则 $\mathrm{d}x=\mathrm{d}u$，
  \begin{align*}
    &=\frac{1}{\sqrt{2\pi}}
      \int f_1(\alpha)\,e^{-\mathrm{i}\omega\alpha}\,\mathrm{d}\alpha
      \int f_2(u)\,e^{-\mathrm{i}\omega u}\,\mathrm{d}u \\
    &=\sqrt{2\pi}\,F_1(\omega)\,F_2(\omega).
  \end{align*}
  $\sqrt{2\pi}$ 来自对称形式两侧各有一个 $1/\sqrt{2\pi}$。
\end{frame}

\begin{frame}{卷积的对称性}
  卷积对两个函数对称：
  \begin{equation*}
    (f*g)(x)=(g*f)(x).
  \end{equation*}
  令 $t=x-\alpha$，则 $\mathrm{d}\alpha=-\mathrm{d}t$，上下限对调后
  \begin{align*}
    (f*g)(x)
    &=\int_{-\infty}^{\infty} f(\alpha)\,g(x-\alpha)\,\mathrm{d}\alpha \\
    &=\int_{-\infty}^{\infty} f(x-t)\,g(t)\,\mathrm{d}t
    =(g*f)(x).
  \end{align*}
  卷积定理也可以写成：卷积的逆变换是像函数之积。对称形式下多出一个 $\sqrt{2\pi}$。
\end{frame}

\begin{frame}{例：两个方波的卷积}
  取两个矩形函数，$a,b>0$，
  \begin{equation*}
    f(x)=\begin{cases}1,&|x|<a,\\0,&\text{其他},\end{cases}
    \qquad
    g(x)=\begin{cases}1,&|x|<b,\\0,&\text{其他}.\end{cases}
  \end{equation*}
  $(f*g)(x)$ 等于区间 $[x-b,x+b]$ 与 $[-a,a]$ 的重叠长度。重叠为空时结果为零，否则
  \begin{equation*}
    (f*g)(x)=
    \begin{cases}
      0, & |x|>a+b,\\
      a+b-|x|, & |a-b|<|x|\le a+b,\\
      \min\{2a,2b\}, & |x|\le |a-b|.
    \end{cases}
  \end{equation*}
  两段斜坡夹一段平台：梯形。若 $a=b$，平台缩成一点，结果是三角波。
\end{frame}

\section{高维傅里叶变换}

\begin{frame}{二维}
  设 $f(x,y)$ 在 $\pm\infty$ 上绝对可积。定义
  \begin{equation*}
    F(k_1,k_2)
    =\frac{1}{2\pi}\iint
      f(x,y)\,e^{-\mathrm{i}(k_1 x+k_2 y)}
      \,\mathrm{d}x\,\mathrm{d}y
  \end{equation*}
  为二维傅里叶变换，并定义
  \begin{equation*}
    f(x,y)
    =\iint
      F(k_1,k_2)\,e^{\mathrm{i}(k_1 x+k_2 y)}
      \,\mathrm{d}k_1\,\mathrm{d}k_2
  \end{equation*}
  为其\textbf{逆变换}。$f$ 与 $F$ 称为\textbf{傅里叶变换对}。
  也可以把系数收成单独的 $1/(2\pi)^2$，结果与上面的形式相应。
\end{frame}

\begin{frame}{矩孔的傅里叶变换}
  求
  \begin{equation*}
    f(x,y)=\begin{cases}
      A, & |x|\leq X,\ |y|\leq Y,\\
      0, & \text{其他}
    \end{cases}
  \end{equation*}
  的傅里叶变换（\textbf{矩孔费琅和夫衍射}）。由上面的定义，
  \begin{equation*}
    F=\frac{A}{2\pi}
      \int_{-X}^{X} e^{-\mathrm{i} k_1 x}\,\mathrm{d}x
      \int_{-Y}^{Y} e^{-\mathrm{i} k_2 y}\,\mathrm{d}y.
  \end{equation*}
  第一个积分
  \begin{align*}
    \int_{-X}^{X} e^{-\mathrm{i} k_1 x}\,\mathrm{d}x
    &=\Bigl[\frac{e^{-\mathrm{i} k_1 x}}{-\mathrm{i} k_1}\Bigr]_{-X}^{X}
     =\frac{e^{\mathrm{i} k_1 X}-e^{-\mathrm{i} k_1 X}}{\mathrm{i} k_1}
     =\frac{2\sin(k_1 X)}{k_1}.
  \end{align*}
  $y$ 方向同样得到 $2\sin(k_2 Y)/k_2$。
\end{frame}

\begin{frame}{例：矩孔的像函数}
  代回，
  \begin{align*}
    F(k_1,k_2)
    &=\frac{A}{2\pi}
      \cdot\frac{2\sin(k_1 X)}{k_1}
      \cdot\frac{2\sin(k_2 Y)}{k_2} \\
    &=\frac{2AXY}{\pi}
      \frac{\sin(k_1 X)}{k_1 X}
      \frac{\sin(k_2 Y)}{k_2 Y}.
  \end{align*}
\end{frame}

  $f(x,y)$ 是孔径面上的复振幅透过率：矩形开口内均匀照明取常值 $A$，挡光处为零。

\begin{frame}{矩孔衍射图样读什么}
  \begin{itemize}
    \item \textbf{中央亮斑}：$k_1=k_2=0$ 时两个 $\sin u/u$ 都取 $1$，
      $F(0,0)=2AXY/\pi$，与孔径面积成正比。
    \item \textbf{暗线}：$k_1 X=n\pi$ 或 $k_2 Y=m\pi$（$n,m=\pm 1,\pm 2,\ldots$）处振幅为零，
      屏上出现互相垂直的暗纹网格。
    \item \textbf{宽窄互换}：孔越窄（$X$ 或 $Y$ 越小），中央峰在对应方向越宽——
      与一维狭缝、高斯函数同一条不确定性。
    \item \textbf{可分离}：$F$ 是 $x$ 方向与 $y$ 方向两个一维狭缝因子的乘积，
      矩孔图样等于两条互相垂直的单缝衍射的叠加结构。
  \end{itemize}
\end{frame}

\begin{frame}{三维}
  $\mathbf{k}=(k_1,k_2,k_3)$，$\mathbf{r}=(x,y,z)$，
  \begin{align*}
    F(\mathbf{k})
      &=\frac{1}{(2\pi)^3}\iiint
        f(\mathbf{r})\,e^{-\mathrm{i}\mathbf{k}\cdot\mathbf{r}}
        \,\mathrm{d}^3\mathbf{r},\\
    f(\mathbf{r})
      &=\iiint
        F(\mathbf{k})\,e^{\mathrm{i}\mathbf{k}\cdot\mathbf{r}}
        \,\mathrm{d}^3\mathbf{k}.
  \end{align*}
  指数可分离时，三重积分拆成三个一维积分。
\end{frame}

\begin{frame}{球对称：化为正弦积分}
  若 $f(\mathbf{r})=f(r)$，$r=|\mathbf{r}|$，则 $F$ 只依赖 $k=|\mathbf{k}|$。
  取极轴沿 $\mathbf{k}$，$\mathbf{k}\cdot\mathbf{r}=kr\cos\theta$，
  \begin{equation*}
    \int_0^{2\pi}\!\mathrm{d}\varphi
    \int_0^{\pi} e^{-\mathrm{i}kr\cos\theta}\sin\theta\,\mathrm{d}\theta
    =\frac{4\pi\sin(kr)}{kr}.
  \end{equation*}
  于是
  \begin{equation*}
    F(k)
    =\frac{1}{2\pi^2 k}
      \int_0^{\infty} r\,f(r)\sin(kr)\,\mathrm{d}r.
  \end{equation*}
  三维球对称傅里叶变换化成一维正弦变换。
\end{frame}

\begin{frame}{例：汤川势（散射常用）}
  取屏蔽库仑（汤川）势
  \begin{equation*}
    f(r)=\frac{e^{-\mu r}}{r},
    \qquad \mu>0.
  \end{equation*}
  用 $\displaystyle\int_0^{\infty}e^{-\mu r}\sin(kr)\,\mathrm{d}r
    =k/(k^2+\mu^2)$，得
  \begin{equation*}
    F(k)=\frac{1}{2\pi^2(k^2+\mu^2)}.
  \end{equation*}
  散射里更常写\textbf{不带} $(2\pi)^{-3}$ 的动量空间矩阵元
  （玻恩近似，$\mathbf{q}=\mathbf{k}'-\mathbf{k}$）：
  \begin{equation*}
    \int\frac{e^{-\mu r}}{r}\,e^{-\mathrm{i}\mathbf{q}\cdot\mathbf{r}}\,\mathrm{d}^3\mathbf{r}
    =\frac{4\pi}{q^2+\mu^2}.
  \end{equation*}
  $\mu\to 0$ 给出库仑势的 $4\pi/q^2$，对应卢瑟福散射振幅的动量依赖。
\end{frame}

\section{$\delta$ 函数}

\begin{frame}{线密度}
  物理学常常要研究一个物理量在空间或时间中分布的密度，例如质量密度、电荷密度、每单位时间传递的动量等等。
  质点、点电荷、瞬时力集中在空间的某一点或时间的某一瞬时，它们的密度又该如何描述？
  \vspace{0.4em}

  若质量 $m$ 均匀分布在长为 $l$ 的线段 $[-l/2,l/2]$ 上，则线密度
  \begin{equation*}
    \rho_l(x)=\frac{m}{l}\,
      \mathrm{rect}\Bigl(\frac{x}{l}\Bigr)
    =\begin{cases}
      m/l, & |x|\leq l/2,\\
      0, & |x|>l/2.
    \end{cases}
  \end{equation*}
\end{frame}

\begin{frame}{先积分，再取极限}
  将 $\rho_l(x)$ 对 $x$ 积分，得到总质量
  \begin{equation*}
    \int_{-\infty}^{\infty}\rho_l(x)\,\mathrm{d}x
    =\int_{-l/2}^{l/2}\frac{m}{l}\,\mathrm{d}x=m.
  \end{equation*}
  让线段长度 $l\to 0$，得到位于坐标原点、质量为 $m$ 的质点。线密度记为 $\rho(x)$，则
  \begin{equation*}
    \lim_{l\to 0}\int_{-\infty}^{\infty}\rho_l(x)\,\mathrm{d}x
    =\int_{-\infty}^{\infty}\rho(x)\,\mathrm{d}x=m.
  \end{equation*}
  若不求积分而先取极限，则
  \begin{equation*}
    \rho(x)=\lim_{l\to 0}\frac{m}{l}\,
      \mathrm{rect}\Bigl(\frac{x}{l}\Bigr)
    =\begin{cases}
      0, & x\neq 0,\\
      \infty, & x=0.
    \end{cases}
  \end{equation*}
  它在 $x=0$ 处为 $\infty$，在 $x\neq 0$ 处为零，它的积分为 $m$。
\end{frame}

\begin{frame}{$\delta$ 函数}
  对于这类集中于一点或一瞬时的模型，引入\textbf{$\delta$ 函数}以描述其密度：
  \begin{equation*}
    \delta(x)=\begin{cases}
      0, & x\neq 0,\\
      \infty, & x=0,
    \end{cases}
    \qquad
    \int_a^b\delta(x)\,\mathrm{d}x
    =\begin{cases}
      1, & a<0<b,\\
      0, & \text{其他情况}.
    \end{cases}
  \end{equation*}
  特别地，
  \begin{equation*}
    \int_{-\infty}^{\infty}\delta(x)\,\mathrm{d}x=1.
  \end{equation*}
  $\delta$ 函数是一种\textbf{广义函数}。用到的是它在积分中的作用。
\end{frame}

\begin{frame}{挑选性}
  作为被积函数的乘积因子，它具有\textbf{挑选性}：
  \begin{equation*}
    \int_{-\infty}^{\infty} f(x)\,\delta(x-a)\,\mathrm{d}x=f(a).
  \end{equation*}
  $f$ 在 $a$ 的邻域里连续时，积分只接收 $x=a$ 处的值。
  \vspace{0.4em}

  $\delta(x)$ 是偶函数，它的导数是奇函数，
  \begin{equation*}
    \delta(-x)=\delta(x),
    \qquad
    \delta'(-x)=-\delta'(x).
  \end{equation*}
  分部积分，边界项在无穷远处为零，
  \begin{equation*}
    \int_{-\infty}^{\infty} f(x)\,\delta'(x-a)\,\mathrm{d}x
    =-f'(a).
  \end{equation*}
\end{frame}

\begin{frame}{自变量是另一个函数}
  设 $\varphi(x_k)=0$，且 $\varphi'(x_k)\neq 0$。在 $x_k$ 附近
  $\varphi(x)\approx\varphi'(x_k)\,(x-x_k)$。由 $\delta(ax)=\delta(x)/|a|$，
  \begin{equation*}
    \delta\bigl[\varphi(x)\bigr]
    =\sum_k\frac{\delta(x-x_k)}{|\varphi'(x_k)|}.
  \end{equation*}
  特例：令 $u=ax$，则 $\mathrm{d}x=\mathrm{d}u/|a|$，
  \begin{equation*}
    \int f(x)\,\delta(ax)\,\mathrm{d}x
    =\int f(u/a)\,\delta(u)\,\frac{\mathrm{d}u}{|a|}
    =\frac{f(0)}{|a|},
  \end{equation*}
  所以 $\delta(ax)=\delta(x)/|a|$。
\end{frame}

\begin{frame}{$\delta(x^2-a^2)$}
  令 $\varphi(x)=x^2-a^2$。实根为 $x=\pm a$，
  $\varphi'(x)=2x$，故 $|\varphi'(\pm a)|=2|a|$。
  \begin{equation*}
    \delta(x^2-a^2)
    =\frac{\delta(x-a)+\delta(x+a)}{2|a|}.
  \end{equation*}
  在这两个根上 $|x|=|a|$，分母也可以写成 $2|x|$。
  \vspace{0.4em}

  例：$\varphi(x)=x^2-4$，根为 $\pm 2$，$|\varphi'|=4$。
  \begin{align*}
    \int_{-\infty}^{\infty}(x^2+1)\,\delta(x^2-4)\,\mathrm{d}x
    &=\frac{2^2+1}{4}+\frac{(-2)^2+1}{4}
     =\frac{5}{2}.
  \end{align*}
\end{frame}

\begin{frame}{三种极限}
  $\delta$ 函数常常可以看作以下函数的极限：
  \begin{align*}
    \delta(x)
      &=\lim_{l\to 0}\frac{1}{l}\,\mathrm{rect}\Bigl(\frac{x}{l}\Bigr),\\
    \delta(x)
      &=\lim_{K\to\infty}\frac{1}{\pi}\frac{\sin(Kx)}{x},\\
    \delta(x)
      &=\lim_{\varepsilon\to 0}
        \frac{1}{\pi}\frac{\varepsilon}{\varepsilon^2+x^2}.
  \end{align*}
  方脉冲的积分是 $1$。后两个的积分分别是
  \begin{equation*}
    \frac{1}{\pi}\int_{-\infty}^{\infty}\frac{\sin u}{u}\,\mathrm{d}u=1,
    \qquad
    \frac{1}{\pi}\bigl[\arctan(x/\varepsilon)\bigr]_{-\infty}^{\infty}=1,
  \end{equation*}
  其中 $u=Kx$。峰越来越窄，积分保持为 $1$。
\end{frame}

\begin{frame}{$\delta$ 函数的傅里叶积分}
  用 $2\pi$ 放在求像函数一侧的约定，
  \begin{equation*}
    C(\omega)
    =\frac{1}{2\pi}\int_{-\infty}^{\infty}
      \delta(x)\,e^{-\mathrm{i}\omega x}\,\mathrm{d}x
    =\frac{1}{2\pi}.
  \end{equation*}
  代回傅里叶积分，
  \begin{equation*}
    \delta(x)=\frac{1}{2\pi}\int_{-\infty}^{\infty}
      e^{\mathrm{i}\omega x}\,\mathrm{d}\omega.
  \end{equation*}
  先积到有限频率 $K$，
  \begin{equation*}
    \frac{1}{2\pi}\int_{-K}^{K} e^{\mathrm{i}\omega x}\,\mathrm{d}\omega
    =\frac{\sin(Kx)}{\pi x},
  \end{equation*}
  再令 $K\to\infty$，回到上一张片子的第二个极限。
\end{frame}

\begin{frame}{多维 $\delta$ 函数}
  $\displaystyle\iiint\delta(\mathbf{r})\,\mathrm{d}x\,\mathrm{d}y\,\mathrm{d}z=1$，
  且 $\mathbf{r}\neq 0$ 时函数值为 $0$。
  直角坐标系中用三个一维 $\delta$ 函数的乘积，
  \begin{equation*}
    \delta(\mathbf{r})=\delta(x)\,\delta(y)\,\delta(z).
  \end{equation*}
  柱坐标的体积元是 $\rho\,\mathrm{d}\rho\,\mathrm{d}\varphi\,\mathrm{d}z$。要保持积分为 $1$，
  \begin{equation*}
    \delta(\mathbf{r})
    =\frac{1}{\rho}\,\delta(\rho)\,\delta(\varphi)\,\delta(z).
  \end{equation*}
  球坐标的体积元是 $r^2\sin\theta\,\mathrm{d}r\,\mathrm{d}\theta\,\mathrm{d}\varphi$，
  \begin{equation*}
    \delta(\mathbf{r})
    =\frac{\delta(r)\,\delta(\theta)\,\delta(\varphi)}{r^2\sin\theta}.
  \end{equation*}
\end{frame}

\begin{frame}{本讲小结：傅里叶变换}
  对称形式
  \begin{equation*}
    F(\omega)=\frac{1}{\sqrt{2\pi}}\int_{-\infty}^{\infty}
      f(x)\,e^{-\mathrm{i}\omega x}\,\mathrm{d}x,
    \qquad
    f(x)=\frac{1}{\sqrt{2\pi}}\int_{-\infty}^{\infty}
      F(\omega)\,e^{\mathrm{i}\omega x}\,\mathrm{d}\omega.
  \end{equation*}
  正余弦形式：$f(x)=\displaystyle\int_0^{\infty}
  \bigl[A(\omega)\cos(\omega x)+B(\omega)\sin(\omega x)\bigr]\,\mathrm{d}\omega$。
  偶函数只有余弦积分，奇函数只有正弦积分。
  \vspace{0.25em}

  典型像函数：
  $e^{-\alpha|x|}\mapsto\sqrt{1/(2\pi)}\,2\alpha/(\alpha^2+\omega^2)$；
  $e^{-\alpha x^2}\mapsto(2\alpha)^{-1/2}e^{-\omega^2/(4\alpha)}$。
  原函数越窄，像函数越宽。
\end{frame}

\begin{frame}{本讲小结：基本定理}
  记 $\mathcal{F}[f]=F$。对称形式下
  \begin{align*}
    \mathcal{F}[f']&=\mathrm{i}\omega\,F,\\
    \mathcal{F}\Bigl[\textstyle\int^{x}f\Bigr]
      &=F/(\mathrm{i}\omega),\\
    \mathcal{F}[f(ax)]
      &=\frac{1}{|a|}F(\omega/a),\\
    \mathcal{F}[f(x-x_0)]
      &=e^{-\mathrm{i}\omega x_0}F,\\
    \mathcal{F}[e^{\mathrm{i}\omega_0 x}f]
      &=F(\omega-\omega_0),\\
    \mathcal{F}[f_1*f_2]
      &=\sqrt{2\pi}\,F_1 F_2.
  \end{align*}
  卷积对称：$f_1*f_2=f_2*f_1$。两方波卷积得梯形；两高斯卷积仍是高斯，方差相加。
\end{frame}

\begin{frame}{本讲小结：$\delta$ 函数}
  \begin{itemize}
    \item 由线密度先积分、再 $l\to 0$ 得到。$x\neq 0$ 时为零，积分为 $1$。
    \item \textbf{挑选性}：$\displaystyle\int f(x)\delta(x-x_0)\,\mathrm{d}x=f(x_0)$。
    \item $\delta(ax)=\delta(x)/|a|$；
      $\displaystyle\delta[\varphi(x)]
      =\sum_i\delta(x-x_i)/|\varphi'(x_i)|$。
    \item 极限表示：方脉冲、$\sin(Kx)/(\pi x)$、
      $\varepsilon/[\pi(\varepsilon^2+x^2)]$。
    \item 对称形式下像函数为常数 $1/\sqrt{2\pi}$：无限局域对应无限平坦的频谱。
    \item 多维：直角坐标 $\delta(\mathbf{r})=\delta(x)\delta(y)\delta(z)$；
      柱、球坐标要除以雅可比因子。
  \end{itemize}
\end{frame}
