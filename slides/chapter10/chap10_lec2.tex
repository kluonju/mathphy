\begin{frame}{本讲提纲}
  边界简单时，格林函数写成自由空间的基本解加上区域外面的像。
  \begin{enumerate}
    \item 三维、二维的基本解；
    \item 接地球和圆盘的反演像；
    \item 四个泊松核：球、半空间、圆盘、半平面；
    \item 圆盘上 $A\cos\phi$ 回到柱坐标的分离变量解。
  \end{enumerate}
\end{frame}

\section{基本解}

\begin{frame}{奇异部分与调和修正}
  \[
    G=G_0+G_1,
    \qquad
    \nabla^2 G_0=\delta(\mathbf{r}-\mathbf{r}_0),
    \qquad
    \nabla^2 G_1=0.
  \]
  狄利克雷问题还要求 $G_1|_{\Sigma}=-G_0|_{\Sigma}$。
  \vspace{0.4em}
  像源放在区域外面。区域内部仍然只有原来的点源，边界条件靠像去配。
\end{frame}

\begin{frame}{三维}
  源在原点时，径向解是 $-C_1/r+C_2$。无穷远处趋于零，则 $C_2=0$。对小球用散度定理，
  \[
    \iint\frac{\partial G_0}{\partial r}\,\mathrm{d}S=4\pi C_1=1,
  \]
  所以 $C_1=1/(4\pi)$。源点移到 $\mathbf{r}_0$，
  \begin{equation*}
    \boxed{\displaystyle
      G_0=-\frac{1}{4\pi|\mathbf{r}-\mathbf{r}_0|}.}
  \end{equation*}
\end{frame}

\begin{frame}{二维}
  同样的手续给出
  \begin{equation*}
    \boxed{\displaystyle
      G_0=\frac{1}{2\pi}\ln|\mathbf{r}-\mathbf{r}_0|.}
  \end{equation*}
  静电里，方程是 $\nabla^2 V=-\rho/\varepsilon_0$ 时，这两个基本解对应自由空间中电荷 $-\varepsilon_0$ 的电位，电荷要改号。
\end{frame}

\section{像}

\begin{frame}{接地球}
  球半径 $a$，源在 $r_0<a$。像放在同一射线上的反演点，
  \[
    \mathbf{r}_1=\frac{a^2}{r_0^2}\mathbf{r}_0,
    \qquad r_1=\frac{a^2}{r_0}>a.
  \]
  球面上 $|\mathbf{r}|=a$ 时，$|\mathbf{r}-\mathbf{r}_1|=(a/r_0)\,|\mathbf{r}-\mathbf{r}_0|$。像的强度取 $a/r_0$，两项相消，
  \begin{equation*}
    G
    =-\frac{1}{4\pi|\mathbf{r}-\mathbf{r}_0|}
     +\frac{a/r_0}{4\pi\bigl|\mathbf{r}-(a^2/r_0^2)\mathbf{r}_0\bigr|}.
  \end{equation*}
  像在球外，球内 $\nabla^2 G$ 只有原来的 $\delta$。
\end{frame}

\begin{frame}{圆盘}
  $\rho<a$ 时用同一个反演点，
  \[
    G
    =\frac{1}{2\pi}\ln|\mathbf{r}-\mathbf{r}_0|
     -\frac{1}{2\pi}\ln|\mathbf{r}-\mathbf{r}_1|
     +\frac{1}{2\pi}\ln\frac{a}{r_0}.
  \]
  常数不破坏调和性。圆周上两个距离相差因子 $a/r_0$，对数相加是零。
\end{frame}

\section{泊松核}

\begin{frame}{球的泊松积分}
  $f=0$ 时只剩 $\partial G/\partial n_0$。在 $r_0=a$ 上像点和源点重合，但法向导数不是零，
  \[
    \frac{\partial G}{\partial r_0}\Big|_{r_0=a}
    =\frac{1}{4\pi a}\frac{a^2-r^2}{R^3},
  \]
  $R^2=r^2+a^2-2ar\cos\Theta$。乘上面积元 $a^2\sin\theta_0\,\mathrm{d}\theta_0\,\mathrm{d}\phi_0$，
  \begin{equation*}
    u
    =\frac{a}{4\pi}\int
      f(\theta_0,\phi_0)
      \frac{a^2-r^2}{(a^2-2ar\cos\Theta+r^2)^{3/2}}
      \,\mathrm{d}\Omega_0.
  \end{equation*}
\end{frame}

\begin{frame}{球心是平均值}
  $r\to a$ 时核变成球面上的 $\delta$，$u$ 回到边界值。
  \vspace{0.4em}
  球心处 $\cos\Theta$ 消失，分母是 $a^3$，分子是 $a^2$，
  \[
    u(0)=\frac{1}{4\pi}\int f\,\mathrm{d}\Omega.
  \]
  调和函数在球心的值等于边界值在球面上的平均。
\end{frame}

\begin{frame}{上半空间}
  像与源反号，放在 $(x_0,y_0,-z_0)$。平面 $z=0$ 上两个距离相等，$G=0$。
  \vspace{0.3em}
  上半空间的外法向是 $-\mathbf{e}_z$。平面上
  \[
    \frac{\partial G}{\partial n_0}
    =\frac{z}{2\pi\bigl[(x-x_0)^2+(y-y_0)^2+z^2\bigr]^{3/2}}.
  \]
  因此
  \[
    u(x,y,z)
    =\frac{z}{2\pi}\iint
      \frac{f(x_0,y_0)}
           {\bigl[(x-x_0)^2+(y-y_0)^2+z^2\bigr]^{3/2}}
      \,\mathrm{d}x_0\,\mathrm{d}y_0.
  \]
\end{frame}

\begin{frame}{圆盘与半平面}
  圆盘上对外法向 $\partial/\partial\rho_0$ 求导，在 $\rho_0=a$ 处
  \[
    u(\rho,\phi)
    =\frac{a^2-\rho^2}{2\pi}\int_0^{2\pi}
      \frac{f(\phi_0)}
           {a^2-2a\rho\cos(\phi-\phi_0)+\rho^2}
      \,\mathrm{d}\phi_0.
  \]
  上半平面用反号像点 $(x_0,-y_0)$，外法向 $-\mathbf{e}_y$，
  \[
    u(x,y)
    =\frac{y}{\pi}\int_{-\infty}^{\infty}
      \frac{f(x_0)}{(x-x_0)^2+y^2}\,\mathrm{d}x_0.
  \]
  $f=1$ 时两个积分都等于 $1$。常值函数调和，边界值也是 $1$。
\end{frame}

\begin{frame}{圆盘上的 $A\cos\phi$}
  边界 $u(a,\phi)=A\cos\phi$。内部有限的解是 $\rho^{n}$ 的傅里叶级数，边界只有 $\cos\phi$，所以
  \[
    u(\rho,\phi)=A\frac{\rho}{a}\cos\phi.
  \]
  这就是柱坐标一章圆内边值的那个函数。泊松积分把分离变量的结果包进一个核对边界值的积分里。
\end{frame}

\begin{frame}{这一讲留下的公式}
  \[
    G_0^{(3)}=-\frac{1}{4\pi R},
    \qquad
    G_0^{(2)}=\frac{1}{2\pi}\ln R.
  \]
  接地球的像在反演点 $r_1=a^2/r_0$，强度是 $a/r_0$。
  \vspace{0.35em}
  球的泊松核是 $(a^2-r^2)/R^3$，半空间是 $z/R^3$，圆盘是 $(a^2-\rho^2)/R^2$，半平面是 $y/R^2$。
\end{frame}
