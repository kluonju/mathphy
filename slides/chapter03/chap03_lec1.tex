\section{泛函}

\subsection{从例子看变分法}

\begin{frame}{两点之间最短路径}
  \textbf{问题}：在平面上给定两点 $(x_1,y_1)$ 与 $(x_2,y_2)$，
  端点固定，找一条连接它们的``最短路径'' $y=u(x)$。
  \begin{itemize}
    \item 常识告诉我们：两点之间线段最短。
    \item 直线方程
      \begin{equation*}
        y = kx + b
          = \frac{y_2 - y_1}{x_2 - x_1}(x - x_1) + y_1.
      \end{equation*}
    \item 这一常识其实在数学上等价于一个\textbf{极小化问题}——
      把``最短''翻译成具体可计算的量。
  \end{itemize}
\end{frame}

\begin{frame}{例 1：弧长作为泛函}
  对任意曲线 $y=u(x)$，由 $ds^2=dx^2+dy^2$ 得弧长
  \begin{equation*}
    J[u] = \int_{x_1}^{x_2}
        \sqrt{1 + u'(x)^2}\,\mathrm{d}x,
  \end{equation*}
  其中 $u'(x) = \mathrm{d}u(x)/\mathrm{d}x$，
  边界条件 $u(x_1)=y_1$，$u(x_2)=y_2$。
  变分问题即求函数 $u(x)$ 使 $J[u]$ 最小。
  这里方括号 $J[u]$ 区别于函数的圆括号，强调``自变量是整个函数''。
\end{frame}

\begin{frame}{例 2：最速降线（捷线）问题}
  \textbf{问题}：质点在重力下沿无摩擦轨道由静止下滑,
  设计轨道形状使下滑时间最短。
  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item 起点 $(x_0,y_0)$，终点 $(x_1,y_1)$，$y$ 轴竖直向上。
        \item 高度 $u$ 处 $v=\sqrt{2g(y_0-u)}$，
          $\mathrm{d}s=\sqrt{1+u'^2}\,\mathrm{d}x$。
        \item 总时间是泛函
          \begin{equation*}
            T[u] = \int_{x_0}^{x_1}
              \frac{\sqrt{1+u'^2}}
                   {\sqrt{2g(y_0-u)}}\,\mathrm{d}x.
          \end{equation*}
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \scalebox{0.58}{\covfig{fastest_tracks}}
      \captionof{figure}{同一对端点上的凹曲线、直线与凸曲线。}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{例 3：最小作用量原理（least action principle）}
  对质量 $m$ 的质点，定义拉格朗日量
  \[
    L = \frac{1}{2}m\dot{\mathbf{r}}^{\,2} - V(\mathbf{r}),
  \]
  与作用量
  \begin{equation*}
    A[\mathbf{r}] = \int_{t_1}^{t_2} L\bigl(\mathbf{r},\dot{\mathbf{r}},t\bigr)\,\mathrm{d}t.
  \end{equation*}
  \textbf{Hamilton 原理}：物理路径使 $A[\mathbf{r}]$ 在端点固定的邻近路径中取极值
  \[
    \delta A = 0.
  \]
  对 $A$ 应用 Euler--Lagrange 方程即得 Newton 第二定律
  $m\ddot{\mathbf{r}} = -\nabla V \equiv \mathbf{F}$。
  变分法不仅处理几何曲线，还能覆盖力学轨迹。
\end{frame}

\begin{frame}{例 4：悬链线问题}
  \textbf{问题}：柔软均匀绳索两端固定，仅受重力作用。
  平衡时绳索呈什么曲线？
  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item 线密度 $\rho$，曲线 $y(x)$ 关于 $y$ 轴对称。
        \item 小段 $\mathrm{d}s=\sqrt{1+y'^2}\,\mathrm{d}x$ 的势能
          $\mathrm{d}U=\rho g\,y\sqrt{1+y'^2}\,\mathrm{d}x$。
        \item 整条绳索势能
          \begin{equation*}
            U[y]=\int_{-L/2}^{L/2}
              \rho g\,y(x)\sqrt{1+y'(x)^2}\,\mathrm{d}x.
          \end{equation*}
        \item 平衡态 = 该势能在长度约束下的极值。
      \end{itemize}
    \end{column}

\end{frame}

\subsection{泛函的定义与记号}

\begin{frame}{泛函的定义}
  \textbf{泛函}：设对于（某一函数集合内的）任意一个函数 $u(x)$，
  有另一个数 $J[u]$ 与之对应，则称 $J[u]$ 为 $u(x)$ 的\textbf{泛函}。
  \begin{itemize}
    \item 自变量：整条函数（整条曲线、轨迹或场）；
    \item 定义域：满足一定边界条件并具有连续二阶导数的\textbf{可取函数}集合；
    \item 直观：``函数 $\to$ 数的映射''，而非常见的``数 $\to$ 数的映射''。
  \end{itemize}
  函数集合的边界条件 + 正则性 = 变分问题的候选集。
\end{frame}

  \begin{columns}[c]

      当红色与蓝色曲线``几乎一致''时，泛函 $J[u]$ 的第一阶变化给出 Euler--Lagrange 方程。

\subsection{泛函的极值与 Euler--Lagrange 方程}

\begin{frame}{泛函的一般形式}
  对于一般情形，函数到数的映射写成
  \begin{equation*}
    J[u] = \int_{a}^{b} \mathrm{d}x\, L(u,u',u'',\cdots\,|\,x),
  \end{equation*}
  其中 $L(u,u',u'',\cdots\,|\,x)$ 为 $u$ 的\textbf{半局域泛函}，
  $u'\equiv\mathrm{d}u/\mathrm{d}x$，$u''\equiv\mathrm{d}^2u/\mathrm{d}x^2$。
  \begin{itemize}
    \item 局域：在 $x$ 处只依赖 $u$ 及该点的有限阶导数。
    \item $L\,\mathrm{d}x$ 是小区间的贡献，积分把整条路径合在一起，称为\textbf{拉格朗日密度}。
  \end{itemize}
\end{frame}

\begin{frame}{变分 $\delta u$ 与 $\delta J$}
  对函数 $u(x)$ 作一微小变化 $\delta u(x)$，称为 $u(x)$ 的\textbf{变分}，
  在积分上下限处 $\delta u(a)=\delta u(b)=0$。
  $J[u]$ 取极值时需满足
  \[
    \delta J[u] \equiv J[u+\delta u] - J[u] = 0.
  \]
  变分与求导可交换次序（$\delta u' = \dfrac{\mathrm{d}}{\mathrm{d}x}\delta u$），
  因此下面的展开能直接做。
\end{frame}

\begin{frame}{Euler--Lagrange 方程的推导}
  对最常见的一阶情形 $L=L(u,u')$，把 $J[u+\delta u]$ 展开到一阶（$fg'=(fg)'-f'g$）：
  \begin{align*}
    J[u+\delta u] &= \int_a^b \mathrm{d}x\, L(u+\delta u,\, u'+\delta u'\,|\,x) \\
    &= \int_a^b \mathrm{d}x \Bigl[L + \frac{\partial L}{\partial u}\delta u
                                  + \frac{\partial L}{\partial u'}\delta u'\Bigr] \\
    &= \int_a^b \mathrm{d}x \Bigl[L + \frac{\partial L}{\partial u}\delta u
       + \frac{\mathrm{d}}{\mathrm{d}x}\!\Bigl(\frac{\partial L}{\partial u'}\delta u\Bigr)
       - \frac{\mathrm{d}}{\mathrm{d}x}\!\Bigl(\frac{\partial L}{\partial u'}\Bigr)\delta u\Bigr] \\
    &= \int_a^b \mathrm{d}x \Bigl[L + \Bigl(\frac{\partial L}{\partial u}
            - \frac{\mathrm{d}}{\mathrm{d}x}\frac{\partial L}{\partial u'}\Bigr)\delta u\Bigr]
       + \Bigl[\frac{\partial L}{\partial u'}\delta u\Bigr]_a^b.
  \end{align*}
  端点条件让末项=0，
  减去 $J[u]$ 后 $\delta J = 0$ 要求
  \begin{equation*}
    \boxed{\frac{\partial L}{\partial u}
    - \frac{\mathrm{d}}{\mathrm{d}x}
      \Bigl(\frac{\partial L}{\partial u'}\Bigr) = 0.}
  \end{equation*}
  这就是\textbf{Euler--Lagrange 方程}。
\end{frame}

\begin{frame}{变分法基本引理}
  如果对所有足够光滑且内部自由的扰动 $\delta u$ 都有
  \begin{equation*}
    \int_a^b \Phi(x)\,\delta u(x)\,\mathrm{d}x = 0,
  \end{equation*}
  则 $\Phi(x)=0$ 在 $[a,b]$ 上逐点成立。
  把 $\delta u$ 取为只在某小段上非零（即\textbf{检验函数}）就足够证明 $\Phi$ 在每一点都为零。
  这就是变分法能给出\textbf{逐点}微分方程的依据。
\end{frame}

\subsection{Euler--Lagrange 方程的应用}

\begin{frame}{例：最短路径是直线}
  对最短路径 $L=\sqrt{1+u'^2}$，偏导
  \[
    \frac{\partial L}{\partial u}=0,\qquad
    \frac{\partial L}{\partial u'}=\frac{u'}{\sqrt{1+u'^2}}.
  \]
  用商法则对第二项沿 $x$ 求全导：
  \begin{align*}
    \frac{\mathrm{d}}{\mathrm{d}x}\frac{u'}{\sqrt{1+u'^2}}
    &= \frac{u''(1+u'^2) - (u')^2 u''}{(1+u'^2)^{3/2}}
     = \frac{u''}{(1+u'^2)^{3/2}}.
  \end{align*}
  代入 Euler--Lagrange 方程，因分母 $(1+u'^2)^{3/2}>0$ 恒正得
  \begin{equation*}
    \frac{u''}{(1+u'^2)^{3/2}}=0
    \;\Longleftrightarrow\;
    u''=0
    \;\Longleftrightarrow\;
    u(x)=kx+b.
  \end{equation*}
  \vspace{0.3em}
  极值曲线是直线，与``两点之间线段最短''一致。
  \hfill\scalebox{0.55}{\covfig{straight_line}}
\end{frame}

\begin{frame}{例：从最小作用量到 Newton 方程}
  设拉格朗日量 $L=\tfrac{1}{2}m\dot{\mathbf{r}}^{\,2}-V(\mathbf{r})$。
  Euler--Lagrange 方程
  \begin{equation*}
    \frac{\mathrm{d}}{\mathrm{d}t}\frac{\partial L}{\partial \dot{\mathbf{r}}}
    - \frac{\partial L}{\partial \mathbf{r}} = 0.
  \end{equation*}
  两条偏导分别为
  \[
    \frac{\partial L}{\partial \dot{\mathbf{r}}} = m\dot{\mathbf{r}},\qquad
    \frac{\partial L}{\partial \mathbf{r}} = -\nabla V(\mathbf{r}).
  \]
  代入得
  \[
    \frac{\mathrm{d}}{\mathrm{d}t}\bigl(m\dot{\mathbf{r}}\bigr)
    + \nabla V(\mathbf{r}) = 0
    \;\Longrightarrow\;
    m\ddot{\mathbf{r}} = -\nabla V \equiv \mathbf{F}.
  \]
  （质量 $m$ 不随时间变化时可移出导数。）
  \textbf{Hamilton 原理为 Newton 力学提供了变分基础。}
\end{frame}

\begin{frame}{最速降线（1/7）：能量守恒首次积分}
  对捷线问题
  \begin{equation*}
    T[u]=\int_{x_0}^{x_1}\frac{\sqrt{1+u'^2}}
                              {\sqrt{2g(y_0-u)}}\,\mathrm{d}x.
  \end{equation*}
  此处 $L=\sqrt{1+u'^2}/\sqrt{2g(y_0-u)}$
  不显含 $x$（自变量），但显含 $u$。
  Euler--Lagrange 方程
  \begin{equation*}
    \frac{\partial L}{\partial u}
   -\frac{\mathrm{d}}{\mathrm{d}x}\frac{\partial L}{\partial u'} = 0
  \end{equation*}
  可以利用 \textbf{Beltrami 恒等式}（或称第一积分）化简。
\end{frame}

\begin{frame}{最速降线（2/7）：Beltrami 恒等式}
  当 $L$ 不显含 $x$ 时,Euler--Lagrange 方程有第一积分
  \begin{equation*}
    L - u'\,\frac{\partial L}{\partial u'} = C \quad(\text{常数}).
  \end{equation*}
  对 $L=\sqrt{1+u'^2}/\sqrt{2g(y_0-u)}$ 计算：
  \begin{align*}
    \frac{\partial L}{\partial u'}
    &= \frac{u'}{\sqrt{1+u'^2}\sqrt{2g(y_0-u)}}, \\
    L - u'\frac{\partial L}{\partial u'}
    &= \frac{1}{\sqrt{2g(y_0-u)}}
       \Bigl(\sqrt{1+u'^2}
            - \frac{u'^2}{\sqrt{1+u'^2}}\Bigr) \\
    &= \frac{1}{\sqrt{2g(y_0-u)}}\cdot
       \frac{1}{\sqrt{1+u'^2}} = C.
  \end{align*}
\end{frame}

\begin{frame}{最速降线（3/7）：摆线方程}
  化简 $(1+u'^2)\cdot 2g(y_0-u) = 1/C^2$。
  令 $\dfrac{1}{C^2} = 2ga$,得到
  \begin{equation*}
    (1+u'^2)\,(y_0-u) = a. \tag{$\ast$}
  \end{equation*}
  即
  \begin{equation*}
    1+u'^2=\frac{a}{y_0-u}.
  \end{equation*}
  左边 $\ge 1$，所以下降高度落在
  $0<y_0-u\le a$ 里。下一步就用一个角把这段高度参数化。
\end{frame}

\begin{frame}{最速降线（4/7）：用角表示下降高度}
  设参数 $t\in(0,\pi)$，令
  \begin{equation*}
    y_0-u=a\sin^2 t. \tag{1}
  \end{equation*}
  代入 $(\ast)$：
  \begin{equation*}
    1+u'^2=\frac{1}{\sin^2 t},
    \qquad
    u'^2=\cot^2 t.
  \end{equation*}
  质点向下滑，$u$ 随 $x$ 减小，故取负号
  \begin{equation*}
    u'=-\cot t. \tag{2}
  \end{equation*}
  由 (1) 直接写出
  \begin{equation*}
    u(t)=y_0-a\sin^2 t
        =y_0-\frac{a}{2}(1-\cos 2t).
  \end{equation*}
  参数 $t$ 在下一步里不再改名。
\end{frame}

\begin{frame}{最速降线（5/7）：积分得摆线}
  由 (1)，$\dfrac{\mathrm{d}u}{\mathrm{d}t}=-2a\sin t\cos t=-a\sin 2t$。
  再由 (2)，$u'=\dfrac{\mathrm{d}u}{\mathrm{d}x}=-\cot t$，故
  \begin{align*}
    \frac{\mathrm{d}x}{\mathrm{d}t}
    &=\frac{\mathrm{d}u/\mathrm{d}t}{\mathrm{d}u/\mathrm{d}x}
      =\frac{-a\sin 2t}{-\cot t}
      =a\cdot 2\sin t\cos t\cdot\tan t \\
    &=2a\sin^2 t
      =a(1-\cos 2t).
  \end{align*}
  积分，并取 $t=0$ 时 $x=x_0$（此时 $u=y_0$，即起点）：
  \begin{equation*}
    x(t)=a\Bigl(t-\frac12\sin 2t\Bigr)+x_0.
  \end{equation*}
  令 $\varphi=2t$、$R=a/2$，即半径为 $R$ 的一拱摆线
  \begin{equation*}
    \boxed{
      \begin{aligned}
        x &= R\,(\varphi-\sin\varphi)+x_0, \\
        u &= y_0-R\,(1-\cos\varphi).
      \end{aligned}
    }
  \end{equation*}
\end{frame}

\begin{frame}{最速降线（6/7）：摆线用时}
  取图中的终点：$\varphi$ 从 $0$ 到 $\pi$（最低点）。
  两端为 $(x_0,y_0)$ 与 $(x_0+\pi R,\,y_0-2R)$。
  由参数式
  \begin{align*}
    \mathrm{d}s
    &= R\sqrt{(1-\cos\varphi)^2+\sin^2\varphi}\,\mathrm{d}\varphi
     = 2R\sin\frac{\varphi}{2}\,\mathrm{d}\varphi, \\
    v
    &= \sqrt{2g\cdot R(1-\cos\varphi)}
     = 2\sin\frac{\varphi}{2}\,\sqrt{gR}.
  \end{align*}
  相除后被积函数是常数：
  \begin{equation*}
    T_{\text{摆线}}
    =\int_0^{\pi}\frac{\mathrm{d}s}{v}
    =\int_0^{\pi}\sqrt{\frac{R}{g}}\,\mathrm{d}\varphi
    =\boxed{\pi\sqrt{\dfrac{R}{g}}}.
  \end{equation*}
\end{frame}

\begin{frame}{最速降线（7/7）：与直线比较}
  连接同一对端点的直线：水平距离 $L=\pi R$，落差 $H=2R$，
  斜率 $u'=-H/L=-2/\pi$。
  沿直线 $y_0-u=(H/L)s=(2/\pi)s$，故
  \begin{align*}
    T_{\text{直线}}
    &=\int_0^{\pi R}
      \frac{\sqrt{1+(2/\pi)^2}}{\sqrt{2g\cdot(2/\pi)\,s}}\,\mathrm{d}s
    =\frac{\sqrt{\pi^2+4}}{\pi}\cdot
      \frac{\sqrt{\pi}}{2\sqrt{g}}\cdot
      2\sqrt{\pi R} \\
    &=\boxed{\sqrt{\pi^2+4}\,\sqrt{\dfrac{R}{g}}}.
  \end{align*}
  两者之比为
  \begin{equation*}
    \frac{T_{\text{直线}}}{T_{\text{摆线}}}
    =\frac{\sqrt{\pi^2+4}}{\pi}\approx 1.185>1.
  \end{equation*}
  摆线用时约为直线的 $84\%$。
\end{frame}

\begin{frame}{最速降线不是直线（图形）}

\subsection{更一般的泛函}

\begin{frame}{高阶泛函（1/2）：变分与一阶展开}
  允许含 $m$ 阶导数时
  \begin{equation*}
    J[u]=\int_a^b L\bigl(x,u,u',\dots,u^{(m)}\bigr)\,\mathrm{d}x.
  \end{equation*}
  对 $u$ 作容许变分 $u\to u+\delta u$（$\delta u$ 任意），
  Taylor 展开到一阶：
  \begin{align*}
    \delta J &\equiv J[u+\delta u]-J[u] \\
    &= \int_a^b \sum_{k=0}^{m}
       \frac{\partial L}{\partial u^{(k)}}\,\delta u^{(k)}\,\mathrm{d}x.
  \end{align*}
  这里用了 $\delta u^{(k)}=\dfrac{\mathrm{d}^k}{\mathrm{d}x^k}\delta u$
  （变分与求导可交换）,
  以及 $\delta x=0$（自变量不变）。
\end{frame}

\begin{frame}{高阶泛函（2/2）：反复分部积分}
  对含 $\delta u^{(k)}$ 的项作 $k$ 次分部积分:
  \begin{align*}
    \int_a^b \frac{\partial L}{\partial u^{(k)}}\delta u^{(k)}\,\mathrm{d}x
    &= \Bigl[\frac{\partial L}{\partial u^{(k)}}\delta u^{(k-1)}\Bigr]_a^b
     - \int_a^b \frac{\mathrm{d}}{\mathrm{d}x}
         \frac{\partial L}{\partial u^{(k)}}\,\delta u^{(k-1)}\,\mathrm{d}x\\
    &= \cdots\\
    &= \Bigl[\text{边界项}\Bigr]_a^b
     +(-1)^k\!\int_a^b
         \frac{\mathrm{d}^k}{\mathrm{d}x^k}\frac{\partial L}{\partial u^{(k)}}\,
         \delta u\,\mathrm{d}x.
  \end{align*}
  对所有 $k$ 求和并要求 $\delta J=0$、$\delta u$ 任意,
  \begin{equation*}
    \boxed{\frac{\delta J}{\delta u}
    = \sum_{k=0}^{m}(-1)^k\frac{\mathrm{d}^k}{\mathrm{d}x^k}
      \Bigl(\frac{\partial L}{\partial u^{(k)}}\Bigr)=0.}
  \end{equation*}
  \textbf{高阶 Euler--Lagrange 方程};$m=1$ 即标准形式。
\end{frame}

\begin{frame}{自然边界条件（1/2）：边界项的形式}
  反复分部积分在 $\delta J$ 中留下的边界项可以整理。
  对 $m$ 阶情形,$\delta J$ 的边界项是
  \begin{equation*}
    \Bigl[\mathcal{B}\Bigr]_a^b,\qquad
    \mathcal{B}=\sum_{j=0}^{m-1}\sum_{k=0}^{j}
        (-1)^k \frac{\mathrm{d}^k}{\mathrm{d}x^k}
        \frac{\partial L}{\partial u^{(k+1)}}\,\delta u^{(j-k)}.
  \end{equation*}
  这是一组关于 $\delta u,\delta u',\dots,\delta u^{(m-1)}$ 的线性组合。
  当 $m=1$ 时只剩一项:
  \[
    \mathcal{B}=\frac{\partial L}{\partial u'}\,\delta u,
  \]
  边界项 $[\mathcal{B}]_a^b$ 即
  $\bigl[\partial L/\partial u'\bigr]_a^b \cdot \delta u$。
\end{frame}

\begin{frame}{自然边界条件（2/2）：分类}
  端点 $a,b$ 处可有三种典型设定:
  \begin{itemize}
    \item \textbf{Dirichlet}:$u(a),u(b)$ 给定,则
      $\delta u(a)=\delta u(b)=0$,边界项自动消失;
    \item \textbf{Neumann}:$u(a),u(b)$ 自由但 $\partial L/\partial u'$ 需为
      $0$:
      \begin{equation*}
        \frac{\partial L}{\partial u'}\Big|_a
        = \frac{\partial L}{\partial u'}\Big|_b = 0;
      \end{equation*}
    \item \textbf{混合}:$u$ 和 $u'$ 部分给定/自由,边界条件混合出现。
  \end{itemize}
  对 $m>1$ 情形,每个高阶导数 $\delta u^{(j)}$ 都对应一组自然条件;
  这是变分原理``多产生''的方程,称为\textbf{自然边界条件}。
\end{frame}

\begin{frame}{多变量与多维情形}
  \textbf{多个因变量}：对 $u_1(x),\dots,u_n(x)$ 各自变分独立，方程为
  \begin{equation*}
    \frac{\partial L}{\partial u_i}
    - \frac{\mathrm{d}}{\mathrm{d}x}\!\Bigl(\frac{\partial L}{\partial u_i'}\Bigr) = 0,
    \quad i=1,\dots,n.
  \end{equation*}
  即使它们由同一拉格朗日密度耦合，每个分量仍然给出自己的方程。
  \textbf{多维场}：对 $u(\mathbf{x})$（$\mathbf{x}\in\Omega$）与
  $J[u]=\int_\Omega F(\mathbf{x},u,\partial_i u)\,\mathrm{d}^d\mathbf{x}$，
  场方程为
  \begin{equation*}
    \frac{\delta J}{\delta u}
    = \frac{\partial F}{\partial u}
      - \partial_i\!\Bigl(\frac{\partial F}{\partial(\partial_i u)}\Bigr) = 0.
  \end{equation*}
  把一维导数换成空间导数 $\partial_i$，形式相同。
\end{frame}

\begin{frame}{例：极小曲面方程（肥皂膜）}
  对二维场 $u(x,y)$ 取 $F=\sqrt{1+u_x^2+u_y^2}$（肥皂膜面积），
  Euler--Lagrange 方程化为
  \begin{equation*}
    \frac{\partial}{\partial x}\!\Bigl(\frac{u_x}{\sqrt{1+u_x^2+u_y^2}}\Bigr)
    +\frac{\partial}{\partial y}\!\Bigl(\frac{u_y}{\sqrt{1+u_x^2+u_y^2}}\Bigr)
    = 0.
  \end{equation*}
  这是\textbf{极小曲面方程}，表达平均曲率为零。
  这就是连续介质力学、静电学、波场理论与场论的统一表达。
\end{frame}

\begin{frame}{例：非局域泛函与 Hartree 项}
  非局域泛函耦合不同点的函数值。例如
  \begin{equation*}
    J[u]=\frac12\iint u(x)K(x,x')u(x')\,\mathrm{d}x\,\mathrm{d}x'.
  \end{equation*}
  当 $K$ 对称时
  \begin{equation*}
    \frac{\delta J}{\delta u(x)} = \int K(x,x')u(x')\,\mathrm{d}x';
  \end{equation*}
  非对称情形下写成 $\frac12\int[K(x,x')+K(x',x)]u(x')\,\mathrm{d}x'$。
  这正是 Hartree 项的形式，将在 DFT 部分再次出现。
\end{frame}

\begin{frame}{例：DFT（1/4）：问题与约束}
  密度泛函理论 (DFT) 把粒子密度 $n(\mathbf{r})$ 作为基本场,
  替代复杂的多体波函数。
  \begin{itemize}
    \item 目标泛函:总能 $E[n]$；
    \item 约束:粒子数守恒
      \begin{equation*}
        \int n(\mathbf{r})\,\mathrm{d}^3\mathbf{r} = N.
      \end{equation*}
  \end{itemize}
  这是\textbf{积分约束下}求泛函极值,处理方式与有限维相同:
  引入 Lagrange 乘子把约束吸收到增广泛函中。
\end{frame}

\begin{frame}{例：DFT（2/4）：增广泛函与 Euler 方程}
  引入\textbf{化学势} $\mu$ 作为粒子数约束的乘子,定义
  \begin{equation*}
    \widetilde E[n,\mu] = E[n] - \mu\Bigl(\int n(\mathbf{r})\,\mathrm{d}^3\mathbf{r} - N\Bigr).
  \end{equation*}
  对 $n(\mathbf{r})$ 求泛函导:
  \begin{equation*}
    \frac{\delta\widetilde E}{\delta n(\mathbf{r})}
    = \frac{\delta E}{\delta n(\mathbf{r})} - \mu.
  \end{equation*}
  令其为零,得到\textbf{DFT 的 Euler 方程}:
  \begin{equation*}
    \boxed{\frac{\delta E}{\delta n(\mathbf{r})} = \mu.}
  \end{equation*}
  平衡时 $\delta E/\delta n$ 在空间各点取值相同,
  等于常数 $\mu$。这正是化学势的物理意义。
\end{frame}

\begin{frame}{例：DFT（3/4）：Thomas--Fermi 模型能量}
  把 $E[n]$ 拆成三类贡献:
  \begin{align*}
    E[n] &= \underbrace{C\int n^{5/3}(\mathbf{r})\,\mathrm{d}^3\mathbf{r}}_{E_{\mathrm{TF}}[n]}
           \;\text{(局域动能)}\\
         &\quad+ \underbrace{\int v_{\mathrm{ext}}(\mathbf{r})\,n(\mathbf{r})\,\mathrm{d}^3\mathbf{r}}_{E_{\mathrm{ext}}[n]}
           \;\text{(外势)}\\
         &\quad+ \underbrace{\frac12\iint
            \frac{n(\mathbf{r})n(\mathbf{r}')}{|\mathbf{r}-\mathbf{r}'|}
            \,\mathrm{d}^3\mathbf{r}\,\mathrm{d}^3\mathbf{r}'}_{E_H[n]}
           \;\text{(Hartree 库仑)} .
  \end{align*}
  $C=\frac{3}{10}(3\pi^2)^{2/3}$ 是 Thomas--Fermi 系数。
  这三项代表了动能、外势、库仑自作用,正好是 DFT 中常见的拆分。
\end{frame}

\begin{frame}{例：DFT（4/4）：逐项求泛函导数}
  对三类贡献分别求 $\delta E/\delta n(\mathbf{r})$:
  \begin{itemize}
    \item \textbf{局域项} $E_{\mathrm{TF}}=\int C\,n^{5/3}\mathrm{d}^3r$:
      \begin{equation*}
        \frac{\delta E_{\mathrm{TF}}}{\delta n(\mathbf{r})}
        = \frac{5}{3}C\,n^{2/3}(\mathbf{r}).
      \end{equation*}
    \item \textbf{线性项} $E_{\mathrm{ext}}=\int v_{\mathrm{ext}}\,n\,\mathrm{d}^3r$:
      \begin{equation*}
        \frac{\delta E_{\mathrm{ext}}}{\delta n(\mathbf{r})}
        = v_{\mathrm{ext}}(\mathbf{r}).
      \end{equation*}
    \item \textbf{非局域项} $E_H$ (Hartree):
      \begin{equation*}
        \frac{\delta E_H}{\delta n(\mathbf{r})}
        = v_H(\mathbf{r})
        = \int \frac{n(\mathbf{r}')}{|\mathbf{r}-\mathbf{r}'|}\,\mathrm{d}^3\mathbf{r}'.
      \end{equation*}
  \end{itemize}
  代入 $\delta E/\delta n = \mu$ 得到 Thomas--Fermi 方程
  \begin{equation*}
    \boxed{\frac{5}{3}C\,n^{2/3}(\mathbf{r}) + v_{\mathrm{ext}}(\mathbf{r}) + v_H(\mathbf{r}) = \mu.}
  \end{equation*}
  局域项与非局域项的耦合清晰:每项贡献自己的泛函导,
  再由粒子数守恒统一锁定 $\mu$。
\end{frame}

\section{哈密顿方程}

\subsection{哈密顿原理与拉格朗日方程}

\begin{frame}{哈密顿原理}
  在经典物理中,拉式量
  $L(q_1,\dots,q_n,\dot q_1,\dots,\dot q_n,t)$
  是广义位置 $q_i$ 与广义速度 $\dot q_i$ 的函数。
  定义\textbf{作用量}(action)
  \begin{equation*}
    S[q] = \int_{t_1}^{t_2} L\bigl(q_i,\dot q_i,t\bigr)\,\mathrm{d}t.
  \end{equation*}
  \textbf{Hamilton 原理}(最小作用量原理):
  物理轨迹是端点固定 ($q_i(t_1),q_i(t_2)$ 给定) 的邻近轨迹中
  使 $S$ 取极值的那一条,即
  \begin{equation*}
    \delta S = 0.
  \end{equation*}
  $\delta S=0$ 不是一个独立假设,而是变分法的直接推论——
  第 1 节已对一般 $J[u]=\int_a^b L(u,u',x)\,\mathrm{d}x$ 推导出
  Euler--Lagrange 方程。把那里的 $x$ 换成 $t$、$u(x)$ 换成 $q_i(t)$,
  立刻得到 $\delta S=0$ 等价于
  \begin{equation*}
    \frac{\mathrm{d}}{\mathrm{d}t}\frac{\partial L}{\partial \dot q_i}
    - \frac{\partial L}{\partial q_i} = 0.
  \end{equation*}
\end{frame}

\begin{frame}{为什么用 Lagrange 形式}
  拉格朗日方程与牛顿运动方程等价，但有一些优点：
  \begin{itemize}
    \item 依赖坐标不必是直角坐标或长度量纲，可根据物理问题选取（角度、约束距离、集体坐标\ldots）；
    \item 对坐标选取具有不变性；
    \item 可扩展到电磁场、量子力学以及规范场论等更广领域；
    \item 对称性与守恒律可通过\textbf{Noether 定理}自然对应。
  \end{itemize}
\end{frame}

\subsection{Legendre 变换与哈密顿方程}

\begin{frame}{正则动量}
  哈密顿注意到拉式运动方程可简化为一系列偏微分方程，称为\textbf{哈密顿方程}。
  首先定义\textbf{正则（广义）动量}
  \begin{equation*}
    p_i = \frac{\partial L}{\partial \dot q_i}.
  \end{equation*}
  对 $L$ 取全微分
  \begin{align*}
    \mathrm{d} L
    &= \sum_i\Bigl(\frac{\partial L}{\partial q_i}\mathrm{d}q_i
                  + \frac{\partial L}{\partial \dot q_i}\mathrm{d}\dot q_i\Bigr)
       + \frac{\partial L}{\partial t}\mathrm{d}t \\
    &= \sum_i\bigl(\dot p_i\,\mathrm{d}q_i + p_i\,\mathrm{d}\dot q_i\bigr)
       + \frac{\partial L}{\partial t}\mathrm{d}t.
  \end{align*}
  （其中第二行用了拉格朗日方程 $\frac{\partial L}{\partial q_i} = \dot p_i$。）
\end{frame}

\begin{frame}{哈密顿量}
  定义体系\textbf{哈密顿量}
  \begin{equation*}
    H(q_i,p_i,t) = \sum_i p_i \dot q_i - L.
  \end{equation*}
  当 $\partial L/\partial \dot q_i$ 可对 $\dot q_i$ 反解时，$H$ 是 $(q,p,t)$ 的函数，
  定义在\textbf{相空间}上。
  取全微分（用 $\mathrm{d}L$ 表达式）：
  \begin{align*}
    \mathrm{d} H
    &= \sum_i\bigl(p_i\,\mathrm{d}\dot q_i + \dot q_i\,\mathrm{d}p_i\bigr)
       - \Bigl[\sum_i\bigl(\dot p_i\,\mathrm{d}q_i + p_i\,\mathrm{d}\dot q_i\bigr)
               + \frac{\partial L}{\partial t}\mathrm{d}t\Bigr] \\
    &= \sum_i\bigl(\dot q_i\,\mathrm{d}p_i - \dot p_i\,\mathrm{d}q_i\bigr)
       - \frac{\partial L}{\partial t}\mathrm{d}t.
  \end{align*}
\end{frame}

\begin{frame}{哈密顿方程}
  另一方面，从定义 $H=H(q_i,p_i,t)$ 又有
  \begin{equation*}
    \mathrm{d} H
    = \sum_i\Bigl(\frac{\partial H}{\partial p_i}\mathrm{d}p_i
                  + \frac{\partial H}{\partial q_i}\mathrm{d}q_i\Bigr)
      + \frac{\partial H}{\partial t}\mathrm{d}t.
  \end{equation*}
  比较 $\mathrm{d}p_i,\mathrm{d}q_i,\mathrm{d}t$ 的系数，立刻得到
  \begin{align*}
    \frac{\partial H}{\partial p_i} &= \dot q_i, \\
    \frac{\partial H}{\partial q_i} &= -\dot p_i, \\
    \frac{\partial H}{\partial t} &= -\frac{\partial L}{\partial t}.
  \end{align*}
  这就是\textbf{Hamilton 正则方程}：两套一阶方程与 Euler--Lagrange 等价，
  但\textbf{坐标与动量对称}。
\end{frame}

\begin{frame}{$H$ 守恒（1/4）：沿轨道取全导数}
  要证明 $H$ 是守恒量,要看 $H$ 沿真实运动轨道
  $q_i(t),\,p_i(t)$ 随 $t$ 的\textbf{全导数}是否为零。
  设 $H=H(q_i,p_i,t)$,则
  \begin{equation*}
    \frac{\mathrm{d}H}{\mathrm{d}t}
    = \sum_i\Bigl(\frac{\partial H}{\partial q_i}\dot q_i
                  + \frac{\partial H}{\partial p_i}\dot p_i\Bigr)
      + \frac{\partial H}{\partial t}.
  \end{equation*}
  这里 $q_i(t),\,p_i(t)$ 满足 Hamilton 方程,
  故我们可以用正则方程替换 $\dot q_i$ 和 $\dot p_i$。
\end{frame}

\begin{frame}{$H$ 守恒（2/4）：代入 Hamilton 方程}
  把 Hamilton 方程
  \begin{equation*}
    \frac{\partial H}{\partial p_i} = \dot q_i,\qquad
    \frac{\partial H}{\partial q_i} = -\dot p_i
  \end{equation*}
  代入全导数式。前两项变成
  \begin{align*}
    \frac{\partial H}{\partial q_i}\dot q_i + \frac{\partial H}{\partial p_i}\dot p_i
    &= (-\dot p_i)\dot q_i + \dot q_i\dot p_i \\
    &= 0.
  \end{align*}
  所以
  \begin{equation*}
    \boxed{\frac{\mathrm{d}H}{\mathrm{d}t}
          = \frac{\partial H}{\partial t}.}
  \end{equation*}
  \textbf{沿真实轨道,$H$ 的全导数等于其偏导数}。
\end{frame}

\begin{frame}{$H$ 守恒（3/4）：与 $L$ 的关系}
  前面得到的 Hamilton 第三方程
  \begin{equation*}
    \frac{\partial H}{\partial t} = -\frac{\partial L}{\partial t}
  \end{equation*}
  给出
  \begin{equation*}
    \frac{\mathrm{d}H}{\mathrm{d}t} = -\frac{\partial L}{\partial t}.
  \end{equation*}
  \textbf{一般结论}：$H$ 沿运动轨道的时间变化率,
  恒等于 $-\partial L/\partial t$。
  \begin{itemize}
    \item 无论 $L$ 是不是动能减势能的标准形式,这条都成立;
    \item 它不要求 $L$ 有任何特殊结构(只要 $H$ 良定义)。
  \end{itemize}
\end{frame}

\begin{frame}{$H$ 守恒（4/4）：守恒条件}
  特别地,如果\textbf{$L$ 不显含时间}
  \begin{equation*}
    \frac{\partial L}{\partial t} = 0,
  \end{equation*}
  那么
  \begin{equation*}
    \frac{\mathrm{d}H}{\mathrm{d}t} = 0
    \quad\Longrightarrow\quad
    H\bigl(q(t),p(t)\bigr) = E = \text{常数}.
  \end{equation*}
  这就是 \textbf{能量守恒}。
  \begin{itemize}
    \item 物理上,$L$ 不显含 $t$ 对应系统的时间平移对称性;
    \item 此时 $H$ 即系统的\textbf{总能量};
    \item 其它守恒律(动量、角动量)由 Noether 定理
      把 $L$ 的空间对称性映射为相应守恒量。
  \end{itemize}
\end{frame}

\begin{frame}{例子：自由粒子与简谐振子}
  \textbf{自由粒子}：$L=\tfrac{1}{2}m\dot q^2$。
  共轭动量 $p=m\dot q$，哈密顿量 $H=p^2/(2m)$。
  Hamilton 方程 $\dot q=p/m$，$\dot p=0$，即匀速运动。
  \vspace{0.3em}
  \textbf{简谐振子}：$L=\tfrac{1}{2}m\dot q^2-\tfrac{1}{2}kq^2$。
  共轭动量 $p=m\dot q$，
  哈密顿量 $H=p^2/(2m)+\tfrac{1}{2}kq^2$。
  Hamilton 方程 $\dot q=p/m$，$\dot p=-kq$，合并即
  $m\ddot q+kq=0$。
\end{frame}

\section{约束问题}

\subsection{有限维 Lagrange 乘子法}

\begin{frame}{约束极值与 Lagrange 乘子}
  我们常常遇到的一类问题是 $u$ 满足其它约束条件。
  以最短路径为例，给定两点 $(x_1,y_1)$、$(x_2,y_2)$，
  增加约束要求曲线所围面积为定值 $A$：
  \begin{equation*}
    \int_{x_1}^{x_2} u(x)\,\mathrm{d}x = A.
  \end{equation*}
  极小化的目标函数变成
  \[
    J[u] = \int_{x_1}^{x_2}\sqrt{1+u'(x)^2}\,\mathrm{d}x
         + \lambda\Bigl(\int_{x_1}^{x_2} u(x)\,\mathrm{d}x - A\Bigr).
  \]
  这里的常数 $\lambda$ 称为\textbf{Lagrange 乘子}——
  它把约束吸收进了极值搜索。
\end{frame}

\begin{frame}{有限维 Lagrange 乘子法的几何}
  把问题暂时退到 $\mathbb{R}^n$：
  设 $f(\mathbf{x})$ 为目标函数，
  $c_a(\mathbf{x})=0\;(a=1,\dots,k)$ 为独立约束。
  引入增广函数
  \begin{equation*}
    \mathcal{F}(\mathbf{x},\lambda_a)
    = f(\mathbf{x}) - \lambda_a c_a(\mathbf{x}).
  \end{equation*}
  把 $\mathbf{x}$ 与 $\lambda_a$ 都视为独立变量，平稳方程是
  \begin{equation*}
    \frac{\partial f}{\partial x_i}
    - \lambda_a\frac{\partial c_a}{\partial x_i} = 0,\qquad
    c_a(\mathbf{x}) = 0.
  \end{equation*}
  对 $\lambda_a$ 求导正好重新得到约束 $c_a=0$。
\end{frame}

\begin{frame}{几何含义}
  $\nabla f$ 给出 $f$ 上升最快的方向，对应 $f$ 等值面的法向。
  在约束曲面上容许的 $\mathrm{d}\mathbf{x}$ 必须满足
  $\nabla c_a\cdot\mathrm{d}\mathbf{x}=0$。
  因此 $\nabla f$ 必须与切空间正交，即
  \[
    \nabla f = \lambda_a\nabla c_a.
  \]
  几何上 $\nabla f$ 是 $\{\nabla c_a\}$ 张成的法向空间中的唯一向量。
  \hfill\scalebox{0.55}{\covfig{lagrange_multipliers}}
\end{frame}

\subsection{多约束与高维例子}

\begin{frame}{例：到平面的最近点（1/2）}
  \textbf{问题}：求平面 $x+y+z=1$ 上到原点的最近点。
  \begin{itemize}
    \item 目标函数（距离平方）：$f(\mathbf{x})=x^2+y^2+z^2$；
    \item 等价约束：$c(\mathbf{x})=x+y+z-1=0$。
  \end{itemize}
  这是一个等式约束的极值问题。
  利用\textbf{Lagrange 乘子法}，定义增广函数
  \begin{equation*}
    \mathcal{F}(\mathbf{x},\lambda)=f(\mathbf{x})-\lambda\,c(\mathbf{x})
    = x^2+y^2+z^2 - \lambda(x+y+z-1).
  \end{equation*}
  把 $\lambda$ 视为独立变量,平稳方程是
  \begin{equation*}
    \nabla_{\mathbf{x},\lambda}\,\mathcal{F}=0.
  \end{equation*}
\end{frame}

\begin{frame}{例：到平面的最近点（2/2）}
  对 $\mathcal{F}$ 各分量求偏导并令其为零:
  \begin{align*}
    \frac{\partial\mathcal{F}}{\partial x}&=2x-\lambda=0,\\
    \frac{\partial\mathcal{F}}{\partial y}&=2y-\lambda=0,\\
    \frac{\partial\mathcal{F}}{\partial z}&=2z-\lambda=0,\\
    \frac{\partial\mathcal{F}}{\partial\lambda}&=-(x+y+z-1)=0.
  \end{align*}
  前三式说明 $2x=2y=2z=\lambda$,即 $x=y=z$。
  代入第四式 $3x=1$ 得 $x=1/3$,从而
  \begin{equation*}
    \boxed{\mathbf{x}^\star=\Bigl(\tfrac13,\tfrac13,\tfrac13\Bigr),\quad
           \lambda=2/3.}
  \end{equation*}
  几何上 $\nabla f=(2/3,2/3,2/3)$ 与法向 $\nabla c=(1,1,1)$ 平行,
  故 $f$ 沿约束面无切向分量,$\mathbf{x}^\star$ 是最近点。
\end{frame}

\begin{frame}{例：球面上的高度极值（1/2）}
  \textbf{问题}：在单位球 $x^2+y^2+z^2=1$ 上求 $f=z$ 的极值。
  \begin{itemize}
    \item 目标：$f(\mathbf{x})=z$；
    \item 约束：$c(\mathbf{x})=x^2+y^2+z^2-1=0$。
  \end{itemize}
  Lagrange 增广函数
  \begin{equation*}
    \mathcal{F}(\mathbf{x},\lambda)=z-\lambda(x^2+y^2+z^2-1).
  \end{equation*}
  对 $(x,y,z,\lambda)$ 求偏导:
  \begin{align*}
    \frac{\partial\mathcal{F}}{\partial x}&=-2\lambda x=0,\\
    \frac{\partial\mathcal{F}}{\partial y}&=-2\lambda y=0,\\
    \frac{\partial\mathcal{F}}{\partial z}&=1-2\lambda z=0,\\
    \frac{\partial\mathcal{F}}{\partial\lambda}&=-(x^2+y^2+z^2-1)=0.
  \end{align*}
\end{frame}

\begin{frame}{例：球面上的高度极值（2/2）}
  联立求解:
  \begin{itemize}
    \item 前两式 $-2\lambda x=0$,$-2\lambda y=0$:
      若 $\lambda\neq 0$ 则 $x=y=0$;
      若 $\lambda=0$ 则第三式 $1=0$ 矛盾,故 $\lambda\neq 0$。
    \item 第三式 $1-2\lambda z=0$ 给出 $z=1/(2\lambda)$。
    \item 第四式约束 $x^2+y^2+z^2=1$:代入 $x=y=0$ 得 $z^2=1$。
  \end{itemize}
  所以 $z=\pm 1$,$\lambda=1/(2z)=\pm 1/2$,极值是
  \begin{equation*}
    \boxed{\mathbf{x}^\star=(0,0,1)\ \text{或}\ (0,0,-1).}
  \end{equation*}
  几何上 $\nabla f=(0,0,1)$ 必须与 $\nabla c=(2x,2y,2z)$ 平行,
  而球面上只有南北两极满足该条件。
\end{frame}

\begin{frame}{例：圆上最大面积多边形（1/3）：面积公式的来源}
  \textbf{问题}：在单位圆上取 $N$ 个顶点,围成的多边形面积最大。
  由 Green 公式,面积等于边界线积分
  \begin{equation*}
    A=\oint_{\partial D}\frac{x\,\mathrm{d}y-y\,\mathrm{d}x}{2}.
  \end{equation*}
  边界是折线。在 $\overline{P_iP_{i+1}}$ 上令
  $x=x_i+t\Delta x$, $y=y_i+t\Delta y$, $t\in[0,1]$,则 $\mathrm{d}x=\Delta x\,\mathrm{d}t$, $\mathrm{d}y=\Delta y\,\mathrm{d}t$,
  \begin{align*}
    x\,\mathrm{d}y-y\,\mathrm{d}x
    &=\bigl[(x_i+t\Delta x)\Delta y-(y_i+t\Delta y)\Delta x\bigr]\mathrm{d}t \\
    &=(x_i\Delta y-y_i\Delta x)\,\mathrm{d}t
     =(x_i y_{i+1}-x_{i+1} y_i)\,\mathrm{d}t.
  \end{align*}
  $t$ 的系数与 $t$ 无关,从 $0$ 积到 $1$ 后再除以 $2$:
  \begin{equation*}
    \int_{\overline{P_iP_{i+1}}}\frac{x\,\mathrm{d}y-y\,\mathrm{d}x}{2}
    =\frac{x_i y_{i+1}-x_{i+1} y_i}{2}.
  \end{equation*}
  右端就是 $\triangle OP_iP_{i+1}$ 的有向面积。
\end{frame}

\begin{frame}{例：圆上最大面积多边形（2/3）：Shoelace 公式}
  \begin{columns}[c]
    \begin{column}{0.56\textwidth}
      各段相加得 \textbf{Shoelace 公式}:
      \begin{equation*}
        2A=\sum_{i=1}^{N}(x_i y_{i+1}-x_{i+1} y_i),
      \end{equation*}
      其中 $(x_{N+1},y_{N+1})\equiv(x_1,y_1)$。
      单位圆上 $x_i=\cos\theta_i$, $y_i=\sin\theta_i$,
      \begin{align*}
        x_i y_{i+1}-x_{i+1} y_i
        &=\sin(\theta_{i+1}-\theta_i).
      \end{align*}
      故目标函数为
      \begin{equation*}
        \boxed{A=\frac12\sum_{i=1}^{N}\sin(\theta_{i+1}-\theta_i).}
      \end{equation*}
      逆时针编号时每一项为正。
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \scalebox{0.92}{\covfig{polygon_circle}}
      \captionof{figure}{阴影三角形面积
        $\tfrac12\sin\Delta\theta$。
        多边形面积是 $N$ 个这样的三角形之和。}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{例：圆上最大面积多边形（3/3）：正多边形}
  顶点已经用角度写出,都在单位圆上。对
  \begin{equation*}
    A=\frac12\sum_{i=1}^{N}\sin(\theta_{i+1}-\theta_i)
  \end{equation*}
  求驻点。$\theta_k$ 只出现在相邻两项里:
  \begin{equation*}
    \frac{\partial A}{\partial\theta_k}
    =\frac12\Bigl[
      \cos(\theta_k-\theta_{k-1})
      -\cos(\theta_{k+1}-\theta_k)
    \Bigr]=0.
  \end{equation*}
  记圆心角 $\alpha_i=\theta_{i+1}-\theta_i\in(0,\pi)$,则
  $\cos\alpha_{k-1}=\cos\alpha_k$,从而 $\alpha_{k-1}=\alpha_k$。
  各角相等,又 $\sum\alpha_i=2\pi$,所以 $\alpha_i=2\pi/N$。
  多边形是\textbf{正 $N$ 边形},由 $N$ 个全等三角形拼成:
  \begin{equation*}
    A_{\max}=\frac{N}{2}\sin\frac{2\pi}{N},
    \quad A_{\max}\xrightarrow{N\to\infty}\pi.
  \end{equation*}
  复数写法放在补充。
\end{frame}

\subsection{积分约束下的变分问题}

\begin{frame}{带积分约束的极值}
  积分约束约束的是整个函数，而不是有限个坐标。
  例：要求
  \[
    \int_a^b u(x)\,\mathrm{d}x = A,
  \]
  同时极值
  \[
    J[u] = \int_a^b L(x,u,u')\,\mathrm{d}x.
  \]
  引入常数乘子 $\lambda$，定义
  \[
    \widetilde J[u] = \int_a^b
      \bigl(L(x,u,u') + \lambda u\bigr)\,\mathrm{d}x - \lambda A.
  \]
  对 $\lambda$ 变分恢复约束，对 $u$ 变分确定形状。
\end{frame}

\begin{frame}{化为普通 Euler--Lagrange 方程}
  $\lambda A$ 是常数项，对 $u$ 变分无影响。
  $\widetilde J$ 的 Euler--Lagrange 方程为
  \begin{equation*}
    \frac{\partial L}{\partial u} + \lambda
    - \frac{\mathrm{d}}{\mathrm{d}x}
      \Bigl(\frac{\partial L}{\partial u'}\Bigr) = 0.
  \end{equation*}
  积分约束只在方程里多出一个常数源项，
  与有限维 Lagrange 乘子法完全平行。
\end{frame}

\begin{frame}{例：给定面积的最短弧}
  在端点固定、$\int u\,\mathrm{d}x=A$ 之下取
  $L=\sqrt{1+u'^2}$。
  Euler--Lagrange 方程为
  \begin{equation*}
    \lambda - \frac{\mathrm{d}}{\mathrm{d}x}
      \Bigl(\frac{u'}{\sqrt{1+u'^2}}\Bigr) = 0.
  \end{equation*}
  第一次积分为
  \begin{equation*}
    \frac{u'}{\sqrt{1+u'^2}} = \lambda x + c.
  \end{equation*}
  \hfill\scalebox{0.55}{\covfig{circle_curve}}
\end{frame}

\begin{frame}{圆弧方程：变量分离与换元}
  从第一次积分 $\dfrac{u'}{\sqrt{1+u'^2}} = \lambda x + c$ 出发，记 $p=\lambda x+c$。
  平方得
  \[
    u' = \frac{p}{\sqrt{1-p^2}}
       = \frac{\lambda x+c}{\sqrt{1-(\lambda x+c)^2}}.
  \]
  换元 $t=\lambda x+c$（$\mathrm{d}t=\lambda\,\mathrm{d}x$）：
  \[
    \mathrm{d}u = u'\,\mathrm{d}x
              = \frac{t}{\sqrt{1-t^2}}\,\frac{\mathrm{d}t}{\lambda}.
  \]
  利用 $\int\frac{t}{\sqrt{1-t^2}}\mathrm{d}t=-\sqrt{1-t^2}$ 积分得
  \[
    u(x) = -\frac{\sqrt{1-(\lambda x+c)^2}}{\lambda} + d.
  \]
  移项再平方，得到圆方程
  \begin{equation*}
    (\lambda x+c)^2 + (\lambda u-d)^2 = 1.
  \end{equation*}
  三个常数 $c,d,\lambda$ 由两个端点条件与面积约束共同定。
\end{frame}

\begin{frame}{几何解读：曲率为常数}
  对第一次积分沿 $x$ 再求一次导：
  \[
    \lambda = \frac{\mathrm{d}}{\mathrm{d}x}\frac{u'}{\sqrt{1+u'^2}}
            = \frac{u''}{(1+u'^2)^{3/2}}.
  \]
  右端恰好是平面曲线 $y=u(x)$ 的\textbf{曲率} $\kappa=\dfrac{u''}{(1+u'^2)^{3/2}}$。
  因此约束下的极值曲线满足
  $\kappa = \lambda = \text{常数}$——
  ``最短弧是圆弧''的几何根源：常曲率曲线只有圆（或退化的直线）。
\end{frame}

\begin{frame}{例：悬链线（1/3）}
  均匀链条线密度 $\rho$，势能
  \[
    U[u] = \rho g \int_{-L/2}^{L/2}
      u(x)\sqrt{1+u'(x)^2}\,\mathrm{d}x.
  \]
  长度约束
  \[
    \int_{-L/2}^{L/2}\sqrt{1+u'(x)^2}\,\mathrm{d}x = \ell.
  \]
  增广拉格朗日量不显含 $x$：
  \begin{equation*}
    F=(\rho g\,u + \lambda)\sqrt{1+u'^2}.
  \end{equation*}
  因此可用 Beltrami 恒等式
  $F-u'\partial F/\partial u'=C$ 求第一次积分。
  \hfill\scalebox{0.40}{\covfig{catenary}}
\end{frame}

\begin{frame}{例：悬链线（2/3）：第一次积分}
  \[
    \frac{\partial F}{\partial u'}
    =(\rho g\,u+\lambda)\frac{u'}{\sqrt{1+u'^2}}.
  \]
  代入 Beltrami 恒等式：
  \begin{align*}
    F-u'\frac{\partial F}{\partial u'}
    &=(\rho g\,u+\lambda)
      \Bigl(\sqrt{1+u'^2}-\frac{u'^2}{\sqrt{1+u'^2}}\Bigr) \\
    &=(\rho g\,u+\lambda)\frac{1}{\sqrt{1+u'^2}}=C.
  \end{align*}
  因而
  \begin{equation*}
    \rho g\,u+\lambda = C\sqrt{1+u'^2},
    \qquad
    u'=\pm\sqrt{\Bigl(\frac{\rho g\,u+\lambda}{C}\Bigr)^2-1}.
  \end{equation*}
\end{frame}

\begin{frame}{例：悬链线（3/3）：积出双曲余弦}
  令 $a=C/(\rho g)$，$w=u+\lambda/(\rho g)$，上式化为
  \[
    \frac{\mathrm{d}w}{\mathrm{d}x}
    =\frac{1}{a}\sqrt{w^2-a^2}.
  \]
  分离变量：
  \[
    \int\frac{a\,\mathrm{d}w}{\sqrt{w^2-a^2}}
    =\int\mathrm{d}x,
    \qquad
    a\,\mathrm{acosh}\frac{w}{a}=x-x_c.
  \]
  代回 $u=w-\lambda/(\rho g)$，并记 $u_c=-\lambda/(\rho g)$：
  \begin{equation*}
    \boxed{u(x)=a\cosh\!\Bigl(\frac{x-x_c}{a}\Bigr)+u_c.}
  \end{equation*}
  $a,x_c,u_c$ 由两端高度与链长 $\ell$ 确定。
  $\cosh z=1+z^2/2+\cdots$，故顶点附近近似为抛物线。
\end{frame}

\begin{frame}{本章总结：主要公式（1/2）}
  \textbf{泛函与 Euler--Lagrange 方程}
  \begin{equation*}
    J[u]=\int_a^b L(x,u,u')\,\mathrm{d}x,
    \qquad
    \frac{\partial L}{\partial u}
    -\frac{\mathrm{d}}{\mathrm{d}x}\frac{\partial L}{\partial u'}=0.
  \end{equation*}
  $L$ 不显含 $x$ 时有 Beltrami 第一积分
  \(L-u'\partial L/\partial u'=C\)。
  \vspace{0.4em}
  \textbf{Hamilton 正则方程}
  \begin{align*}
    p_i&=\frac{\partial L}{\partial\dot q_i},
    &
    H&=\sum_i p_i\dot q_i-L,\\
    \dot q_i&=\frac{\partial H}{\partial p_i},
    &
    \dot p_i&=-\frac{\partial H}{\partial q_i}.
  \end{align*}
  $\partial L/\partial t=0$ 时 $H=E$ 守恒。
\end{frame}

\begin{frame}{本章总结：主要公式（2/2）}
  \textbf{约束}：增广目标 $\mathcal{F}=f-\lambda_a c_a$，
  平稳条件 $\partial f/\partial x_i=\lambda_a\partial c_a/\partial x_i$。
  积分约束 $\int u\,\mathrm{d}x=A$ 相当于把 $L$ 换成 $L+\lambda u$。
  \vspace{0.5em}
  \textbf{两条经典极值曲线}
  \begin{align*}

    &u=a\cosh\!\Bigl(\frac{x-x_c}{a}\Bigr)+u_c.
  \end{align*}
\end{frame}

\begin{frame}{补充：圆上多边形的复数表示}
  令 $z_i = x_i + \mathrm{i} y_i$, $\bar z_i = x_i - \mathrm{i} y_i$。
  在单位圆上 $\bar z_i = 1/z_i$。
  \vspace{0.2em}
  直接展开可验证
  \begin{align*}
    x_i y_{i+1} - x_{i+1} y_i
    &= \tfrac{1}{2\mathrm{i}}\bigl(z_i \bar z_{i+1} - \bar z_i z_{i+1}\bigr).
  \end{align*}
  因此面积写成
  \begin{equation*}
    \boxed{2A = \frac{1}{2\mathrm{i}}\sum_{i=1}^{N}
          \bigl(z_i \bar z_{i+1} - \bar z_i z_{i+1}\bigr)
        = \text{Im}\sum_{i=1}^{N}\bar z_i z_{i+1}.}
  \end{equation*}
  若 $z_i = \mathrm{e}^{\mathrm{i}\theta_i}$,$\bar z_i z_{i+1} = \mathrm{e}^{\mathrm{i}(\theta_{i+1}-\theta_i)}$,
  \begin{equation*}
    2A = \sum_{i=1}^{N}\sin(\theta_{i+1}-\theta_i).
  \end{equation*}
  几何上:每段对应的有向面积即相邻两点的张角的正弦。
\end{frame}

\begin{frame}{补充：复数形式的变分方程}
  在复数下写出约束 $c_i=|z_i|^2-1=0$,
  增广泛函 (乘子 $\lambda_i$)
  \begin{equation*}
    \mathcal{F}=2A-\sum_{i=1}^{N}\lambda_i(|z_i|^2-1).
  \end{equation*}
  把 $z_i,\bar z_i$ 视为独立变量,得
  \begin{align*}
    \frac{\partial\mathcal{F}}{\partial \bar z_i}
    &= z_{i+1}-z_{i-1} - 2\lambda_i z_i = 0,\\
    \frac{\partial\mathcal{F}}{\partial z_i}
    &= \bar z_{i+1}-\bar z_{i-1} - 2\lambda_i \bar z_i = 0.
  \end{align*}
  第二式是第一式的复共轭,故只取第一式
  \begin{equation*}
    \boxed{z_{i+1}-z_{i-1} = 2\lambda_i z_i.}
  \end{equation*}
  这相当于实部情形下两条方程 $y_{i+1}-y_{i-1}=2\lambda_i x_i$,
  $-(x_{i+1}-x_{i-1})=2\lambda_i y_i$ 的复数合并写法。
\end{frame}

\begin{frame}{补充：由复数方程得到正多边形}
  把 $z_i=\mathrm{e}^{\mathrm{i}\theta_i}$ 代入 $z_{i+1}-z_{i-1}=2\lambda_i z_i$,
  即
  \begin{equation*}
    \mathrm{e}^{\mathrm{i}\theta_{i+1}}-\mathrm{e}^{\mathrm{i}\theta_{i-1}}
    = 2\lambda_i \mathrm{e}^{\mathrm{i}\theta_i}.
  \end{equation*}
  设相邻差为常数 $\alpha=\theta_{i+1}-\theta_i$,则
  $\theta_{i+1}-\theta_{i-1}=2\alpha$,等式化为
  \begin{equation*}
    \mathrm{e}^{\mathrm{i}\alpha}-\mathrm{e}^{-\mathrm{i}\alpha}=2\lambda_i
    \;\Longrightarrow\;
    \lambda_i = \mathrm{i}\sin\alpha\ (\text{与 }i\text{ 无关}).
  \end{equation*}
  各 $\lambda_i$ 相同说明平稳解\textbf{等角间距}。
  由 $\sum_{i=1}^N\alpha=2\pi$ 得 $\alpha=2\pi/N$,
  故多边形是\textbf{正 $N$ 边形},
  $A_{\max}=\dfrac{N}{2}\sin\dfrac{2\pi}{N}$。
\end{frame}
