\begin{hwproblem}{1}
  在 $x_0=0$ 的邻域上求解高斯方程 (超几何级数微分方程) 
  $$x(x-1) y^{\prime \prime}+[(1+\alpha+\beta) x-\gamma] y^{\prime}+\alpha \beta y=0,$$
  其中$\alpha, \beta, \gamma$为常数.
\end{hwproblem}

\begin{hwproblem}{3}
对于二阶常系数线性齐次方程, $$
y'' + a y' + by = 0,
$$
其特征根为二重实根$m$.
利用朗斯基行列式的概念推导其解为 
\begin{equation*}
    y = (C_1 + C_2  x) e^{m x}.
\end{equation*}
\end{hwproblem}
