\begin{hwproblem}{1}
  写出下列复数的实部,虚部,模和辐角.
  \begin{enumerate}
    \item $1 + \ii \sqrt{3}$,
    \item $\frac{1-\ii}{1+\ii}$,
    \item $e^{z}$,
    \item $e^{1+\ii}$,
    \item $e^{\ii \varphi(x)}$, $\varphi(x)$是实数$x$的实函数.
  \end{enumerate}
\end{hwproblem}

\vspace{0.5cm}

\begin{hwproblem}{2}
  证明以下规律:
  \begin{enumerate}
    \item 复数的加法和乘法满足结合律,即 
    \begin{itemize}
      \item $(z_1 + z_2 ) + z_3 = z_1 +( z_2 + z_3)$
      \item $(z_1 \cdot z_2) \cdot z_3 = z_1 \cdot (z_2 \cdot z_3)$
    \end{itemize}

    \item 复数乘积的共轭等于复数共轭的乘积,即$(z_1 \cdot z_2)^* = z_1^* z_2^*$.
  \end{enumerate}  
\end{hwproblem}

\vspace{0.5cm}

\begin{hwproblem}{3}
  找出复平面上满足方程\[|z - \ii a|  = \lambda | z + \ii a|, \lambda > 0,\]
  的所有点$z = x + \ii y$.请分$\lambda$的三种情况(1) $\lambda<1$, (2) $\lambda >1$,(3) $\lambda = 1$分别绘制图形.
\end{hwproblem}

\vspace{0.5cm}

\begin{hwproblem}{4}
  若$|z|=1$, $a,b$为任意复数,试证明
  $$
  \left| \frac{a z + b}{\bar{b} z + \bar{a}} \right| = 1 .
  $$
\end{hwproblem}

\begin{hwproblem}{5}

\begin{hwproblem}{6}
  求下列方程的根,并在复平面上画出它们的位置.
  \begin{enumerate}
    \item $z^4 + 1 = 0$,
    \item $z^2 + 2 z \cos \lambda + 1 = 0, \quad 0 < \lambda < \pi$.
  \end{enumerate}
\end{hwproblem}

\begin{hwproblem}{7}

$$
\tan^{-1}(z)=\frac{\ii }{2}[\ln (1-\ii z)-\ln (1+\ii z)] .

\end{hwproblem}

\begin{hwproblem}{8}
  试证明极坐标下的柯西-黎曼方程
  $$
    \left\{\begin{array}{l}
    \frac{\partial u}{\partial \rho}=\frac{1}{\rho} \frac{\partial v}{\partial \varphi} ,\\
    \frac{1}{\rho} \frac{\partial u}{\partial \varphi}=-\frac{\partial v}{\partial \rho} .
    \end{array}\right.
  $$
\end{hwproblem}
