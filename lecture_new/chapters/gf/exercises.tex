\section{Exercises}

\begin{enumerate}
\item Using the method of images, solve the Dirichlet problem for Laplace's equation in the disk $\rho<a$ with $u|_{\rho=a}=f(\varphi)$. Recover the Poisson integral formula of the text.

\item Solve $\nabla^2 u=0$ in the half-plane $y>0$ with $u|_{y=0}=f(x)$ by the method of images.

\item For the disk $\rho\le a$, find the harmonic function with boundary values
\begin{enumerate}
\item $u|_{\rho=a}=A\cos\varphi$;
\item $u|_{\rho=a}=A+B\sin\varphi$.
\end{enumerate}

\item Expand the kernel of the disk Poisson integral in a Fourier series, integrate term by term, and compare with the separation-of-variables solution.

\item Construct the Dirichlet Green's function for the infinite slab $0<z<H$.

\item A string fixed at both ends is driven by $\rho f=\rho\Phi(x)\sin\omega t$. Solve the forced problem, discuss resonance, and write the resonant solution when it occurs.

\item The same string is driven by a concentrated harmonic force $\rho f_0\sin\omega t$ at an interior point $x_0$. Find the displacement.

\item A thin wire of length $\ell$ and resistance density $R$ carries current $I$ and exchanges heat with a medium at zero temperature. With vanishing initial and boundary data for the temperature, set up and solve the problem by the impulse method.

\item Solve $u_t-a^2 u_{xx}=0$ for $x>0$ with $u|_{x=0}=At$ and $u|_{t=0}=0$.

\item Solve $u_t-a^2 u_{xx}=0$ for $x>0$ with $u|_{x=0}=f(t)$ and $u|_{t=0}=\phi(x)$.

\item Solve $u_t-a^2 u_{xx}=0$ for $x>0$ with $u_x|_{x=0}=q(t)$ and $u|_{t=0}=0$.

\item Using the generalized Green formula, solve
\begin{equation}
u_{tt}-a^2 u_{xx}=f(x,t),
\qquad
u\big|_{t=0}=\varphi(x),
\qquad
u_t\big|_{t=0}=\psi(x).
\end{equation}

\item Solve $x^2 u_{tt}-t^2 u_{xx}=0$ with $u|_{t=1}=\varphi(x)$ and $u_t|_{t=1}=\psi(x)$ by means of a Riemann function.
\end{enumerate}
