\section{Applications of integral transforms}

The three calculations in this section are the standard uses of the Laplace transform once the operational rules are available: a series is turned into an integral, an integral containing $1/t$ is turned into an integral of a transform, and a linear initial-value problem is turned into algebra. Each one uses a transform pair already derived.

\subsection{A series as a Laplace integral}

From $\mathcal{L}\{t\}=1/p^2$ with $\operatorname{Re}p>0$, the substitution $p=n$ gives
\begin{equation}
\frac{1}{n^2}=\int_0^{\infty}t e^{-nt}\,\mathrm{d}t,
\qquad n=1,2,3,\ldots.
\end{equation}
Summing the geometric series inside the integral is legitimate because the summand is nonnegative:
\begin{equation}
\sum_{n=1}^{\infty}\frac{1}{n^2}
=\int_0^{\infty}t\sum_{n=1}^{\infty}e^{-nt}\,\mathrm{d}t
=\int_0^{\infty}t\cdot\frac{e^{-t}}{1-e^{-t}}\,\mathrm{d}t
=\int_0^{\infty}\frac{t}{e^t-1}\,\mathrm{d}t.
\end{equation}
The integral equals $\pi^2/6$. That value was already obtained in \eqref{eq:ch05-basel} by applying Parseval's identity to $f(x)=x$ on $[-\pi,\pi]$, whose nonzero Fourier coefficients satisfy $|c_n|=1/|n|$. The two derivations check each other: the series is the mean square of a sawtooth, and it is also the integral of $t/(e^t-1)$.

\subsection{Integrals that contain a factor $1/t$}

Suppose $\int_p^{\infty}\bar f(q)\,\mathrm{d}q$ exists and $f(t)/t$ remains bounded as $t\to 0$. Integrating the Laplace integral in the parameter produces
\begin{equation}
\int_p^{\infty}\bar f(q)\,\mathrm{d}q
=\int_0^{\infty}\frac{f(t)}{t}e^{-pt}\,\mathrm{d}t,
\end{equation}
which is the statement
\begin{equation}
\int_p^{\infty}\bar f(q)\,\mathrm{d}q\longleftrightarrow\frac{f(t)}{t}.
\label{eq:ch06-one-over-t}
\end{equation}
Send $p\to 0$ from the right, when the resulting integrals converge:
\begin{equation}
\int_0^{\infty}\bar f(p)\,\mathrm{d}p=\int_0^{\infty}\frac{f(t)}{t}\,\mathrm{d}t.
\label{eq:ch06-integral-zero}
\end{equation}

\begin{example}[The Dirichlet integral]
Take $f(t)=\sin t$, so $\bar f(p)=1/(p^2+1)$. Then \eqref{eq:ch06-integral-zero} gives
\begin{equation}
\int_0^{\infty}\frac{\sin t}{t}\,\mathrm{d}t
=\int_0^{\infty}\frac{1}{p^2+1}\,\mathrm{d}p
=\frac{\pi}{2}.
\end{equation}
The same identity with a general frequency reads
\begin{equation}
\int_p^{\infty}\frac{\omega}{q^2+\omega^2}\,\mathrm{d}q
=\frac{\pi}{2}-\arctan\Bigl(\frac{p}{\omega}\Bigr)
\longleftrightarrow\frac{\sin(\omega t)}{t},
\end{equation}
and the case $p\to 0$ recovers $\int_0^{\infty}(\sin\omega t)/t\,\mathrm{d}t=\pi/2$ for $\omega>0$.
\end{example}

\begin{example}[A difference of cosines]
For $a>0$ and $b>0$ the integral $\int_0^{\infty}(\cos(at)-\cos(bt))/t\,\mathrm{d}t$ is not a residue calculation on a semicircle in any very direct way, because the integrand is even only after a careful limiting argument and decays slowly. The transform route is short. Here
\begin{equation}
\cos(at)-\cos(bt)
\longleftrightarrow
\frac{p}{p^2+a^2}-\frac{p}{p^2+b^2},
\end{equation}
and \eqref{eq:ch06-integral-zero} gives
\begin{align}
\int_0^{\infty}\frac{\cos(at)-\cos(bt)}{t}\,\mathrm{d}t
&=\int_0^{\infty}
\left(\frac{p}{p^2+a^2}-\frac{p}{p^2+b^2}\right)\mathrm{d}p
\nonumber\\
&=\frac12\Bigl[\ln\frac{p^2+a^2}{p^2+b^2}\Bigr]_0^{\infty}
=\ln\frac{b}{a}.
\end{align}
The boundary term at infinity vanishes, and the term at zero is $-\ln(a/b)$.
\end{example}

\subsection{An initial-value problem}

\begin{example}[A harmonic oscillator released from rest]
A mass $m$ on a spring of stiffness $k$ satisfies
\begin{equation}
mX''(t)+kX(t)=0,
\qquad X(0)=X_0,\quad X'(0)=0.
\end{equation}
Let $x(p)=\mathcal{L}\{X(t)\}$ and $\omega_0^2=k/m$. The second-derivative rule \eqref{eq:ch06-laplace-nth} contributes $p^2x-pX_0$, because $X'(0)=0$. The transformed equation is
\begin{equation}
m\bigl(p^2x(p)-pX_0\bigr)+kx(p)=0,
\end{equation}
so
\begin{equation}
x(p)=X_0\frac{p}{p^2+\omega_0^2}.
\end{equation}
The cosine pair inverts this at once:
\begin{equation}
X(t)=X_0\cos(\omega_0 t).
\end{equation}
The mass starts at $X_0$ with zero velocity, and it oscillates at the natural frequency. No homogeneous solution has to be fitted afterwards: both initial conditions were used when the derivative rule was applied.
\end{example}

\begin{example}[The same oscillator with a nonzero initial velocity]
If instead $X'(0)=V_0$, the derivative rule contributes an additional $-mV_0$. Then
\begin{equation}
\bigl(p^2+\omega_0^2\bigr)x(p)=pX_0+V_0,
\end{equation}
and
\begin{equation}
x(p)=X_0\frac{p}{p^2+\omega_0^2}+\frac{V_0}{p^2+\omega_0^2}.
\end{equation}
Inversion gives
\begin{equation}
X(t)=X_0\cos(\omega_0 t)+\frac{V_0}{\omega_0}\sin(\omega_0 t).
\end{equation}
The coefficient of the sine is $V_0/\omega_0$ rather than $V_0$, because $\mathcal{L}\{\sin(\omega_0 t)\}=\omega_0/(p^2+\omega_0^2)$. Differentiating the solution at $t=0$ recovers $V_0$ and is the quickest check of that factor.
\end{example}
